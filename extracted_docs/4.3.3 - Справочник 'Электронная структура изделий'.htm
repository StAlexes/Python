Справочник "Электронная структура изделий"
  Внимание

Данный раздел предназначен для описания работы с устаревшим механизмом, основанном на взаимодействии справочника ЭСИ со связанными справочниками (Документы, Материалы, Электронные компоненты и т.д.).
На текущий момент работа с объектами справочника ЭСИ не отличается от работы с объектами других справочников, поддерживающих историю изменений. Подробнее о работе с объектами: «Создание объектов справочника»

Справочник "Электронная структура изделий" предназначен для управления структурами изделий и ведения номенклатуры предприятия.

C#
Копировать 
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Documents;
using TFlex.DOCs.Model.References.Nomenclature;
using TFlex.DOCs.Model.Structure;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        var documentReference = new DocumentReference(Context.Connection); // Создаем экземпляр справочника 'Документы'
        var document = documentReference.FindOne(documentReference.ParameterGroup[SystemParameterType.ObjectId], 12); // Находим документ по его идентификатору
        if (document is null)
            return;

        var nomenclatureReference = new NomenclatureReference(Context.Connection); // Создаем экземпляр справочника 'Электронная структура изделий'

        NomenclatureObject nomenclatureObject = nomenclatureReference.FindByLinkedObject(document); // Находим электронную структуру по связанному объекту
        if (nomenclatureObject is null)
        {
            NomenclatureObject nomenclature = nomenclatureReference.CreateNomenclatureObject(document); // Создаем электронную структуру по связанному объекту
            // Фиксируем изменения
            nomenclature.EndChanges();
            nomenclature.CheckIn("Применение изменений");
        }
    }
}

Получение связанных файлов объекта ЭСИ
C#
Копировать 
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Files;
using TFlex.DOCs.Model.References.Nomenclature;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        if (Context.ReferenceObject is not NomenclatureObject nomenclatureObject) // Проверяем, что текущий объект является объектом ЭСИ
            return;

        List<FileObject> files = nomenclatureObject.GetObjects<FileObject>(NomenclatureReferenceObject.RelationKeys.Files); // Получаем связанные файлы
    }
}
