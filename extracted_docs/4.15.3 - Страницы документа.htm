Страницы документа

В данном примере показано как получить информацию о страницах документа системы T-FLEX CAD (*.grb)

C#
Копировать 
using TFlex.DOCs.Model.FilePreview.CADService;
using TFlex.DOCs.Model.FilePreview.CADService.TFlexCadDocument;
using TFlex.DOCs.Model.Macros;

public class GetPages : MacroProvider
{
    public GetPages(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        string pathToGrb = "Путь к grb-файлу";
        CadDocumentProvider provider = CadDocumentProvider.Connect(Context.Connection, ".grb");  // Подключаемся к CAD-системе
        using CadDocument document = provider.OpenDocument(pathToGrb, true); // Открываем документ в режиме чтения

        if (document is null)
            Error("Ошибка", $"Файл '{pathToGrb}' не может быть открыт");

        TFlexPageInfo[] tflexPages = document.GetTFlexPagesInfo(new TFlexPageType[] { TFlexPageType.Normal, TFlexPageType.BillOfMaterials }); // Получаем страницы типа "Обычная" и "Спецификация"

        foreach (TFlexPageInfo page in tflexPages)
        {
            TFlexPageType type = page.PageType; // Тип страницы
            string name = page.Name; // Наименование
            int index = page.Index; // Индекс
            bool visible = page.Visible; // Видимость
            string format = page.Properties.Paper.Format; // Формат листа
            PaperOrientation paperOrientation = page.Properties.Paper.Orientation; // Ориентация листа
        }

        document.Close(false); // Закрываем документ без сохранения
    }
}
