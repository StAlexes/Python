Изменение актуальной версии объекта справочника

В данном примере рассмотрен вопрос, как сделать актуальной указанную версию объекта.

Пример
C#
Копировать 
using System;
using System.Linq;
using TFlex.DOCs.Model.Desktop;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class ChangeCurrentVersionReferenceObject : MacroProvider
{
    // Название искомой версии объекта
    private readonly string versionName = "name";

    public ChangeCurrentVersionReferenceObject(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Объект, текущую версию которого заменяем более ранней
        ReferenceObject referenceObject = Context.ReferenceObject;

        if (referenceObject == null || !referenceObject.CanCheckOut)
            return;

        // Получаем список именованных изменений объекта справочника. Если необходимо получить все изменения, то вторым аргументом необходимо передать false
        var changes = Desktop.GetObjectChangelists(referenceObject, true);

        // Осуществляем поиск изменения с указанной версией 
        var change = changes.FirstOrDefault(ch => String.Equals(ch.Changelist.Label, versionName, StringComparison.OrdinalIgnoreCase));

        if (change == null)
        {
            Сообщение("Внимание!", "Не найдена указанная версия объекта");
            return;
        }

        // Делаем актуальной указанную версию объекта
        ReferenceObject newActualObject = Desktop.CheckOutVersion(referenceObject, change.Version);
        // Применяем изменения
        Desktop.CheckIn(newActualObject, "Изменили актуальную версию объекта", false);
    }
}
