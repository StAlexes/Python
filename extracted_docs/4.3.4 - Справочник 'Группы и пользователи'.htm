Справочник "Группы и пользователи"

Справочник "Группы и пользователи" предназначен для управления параметрами учетных записей системы и содержит сведения о пользователях.

C#
Копировать 
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        var userReference = new UserReference(Context.Connection); // Создаем экземпляр справочника

        List<User> allUsers = userReference.GetAllUsers(); // Получаем всех пользователей

        User employee = userReference.FindUser("Сотрудник"); // Находим пользователя по полному имени
        employee.LockAccount(); // Отключаем учетную запись
    }
}
