using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Mail;
using TFlex.DOCs.UI.Client.Common.ViewModels;
using TFlex.DOCs.UI.Client.Common.ViewModels.Mail;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public string GetAttachments()
    {
        if (Context is not UIMacroContext uIMacroContext) // Получаем UI контекст выполнения макроса
            return String.Empty;

        // Из контекста получаем модель представления текущего почтового элемента, если это не сообщение/задание, выходим
        if (uIMacroContext.ObjectViewModel is not MailItemViewModel mailMessageVm)
            return String.Empty;

        MailItem mailItem = mailMessageVm.Model; // Получаем почтовый элемент
        AttachmentCollection attachments = mailItem.Attachments; // Получаем вложения

        // Если необходимо проверить, является ли текущий почтовый элемент сообщением и обработать его
        //if (mailItem is MailMessage message)
        //{
        //    // ...
        //}

        // Если необходимо обработать текущее задание
        //if (mailItem is MailTask task)
        //{
        //    // ...
        //}

        return String.Join("; ", attachments); // Возвращаем вложения в строковом представлении
    }
}
