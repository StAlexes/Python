Пакетная обработка

Класс Reference предоставляет методы пакетной обработки объектов, позволяющих увеличить скорость выполнения операций и уменьшить количество запросов к серверу.

Перезагрузка объектов

Reference.Reload("Список объектов") - перезагружает параметры и связи объектов

Reference.Reload("Список объектов"[, "Список настроек загрузки объектов", "Загружать удалённые объекты"]) - перезагружает параметры и связи объектов с настройками дополнительных параметров для загрузки. Аргумент "Загружать удалённые объекты" отвечает за загрузку удалённых объектов и их учитывание при расчёте значений параметров. С информацией о настройке дополнительных параметров для загрузки можно ознакомиться в разделе «Оптимизация загрузки объектов»

Пример
Копировать 
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        ReferenceObject[] selectedObjects = Context.GetSelectedObjects(); // Получаем выбранные объекты справочника

        Reference.Reload(selectedObjects); // Перезагружаем объекты
    }
}

 

Загрузка подписей объектов

Reference.LoadSignatures("(Список объектов)") - производит загрузку подписей объектов справочников

Пример
Копировать 
using System;
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // В коде необходимо пустые значения глобальных идентификаторов (Guid.Empty) заменить на реальные

        ReferenceObject[] referenceObjectList1 = Context.GetSelectedObjects(); // Получаем выбранные объекты справочника

        // Для примера загрузим объекты еще одного справочника
        var reference = Context.Connection.ReferenceCatalog.Find(Guid.Empty).CreateReference();
        using var filter = new Filter(reference.ParameterGroup); // Создаем фильтр для поиска объектов во втором справочнике
        filter.Terms.AddTerm(reference.ParameterGroup[SystemParameterType.ObjectId], ComparisonOperator.GreaterThan, 5);
        List<ReferenceObject> referenceObjectList2 = reference.Find(filter); // Находим объекты второго справочника

        // Пакетная загрузка подписей работает с объектами разных справочников.
        // При загрузке подписей пропускаются объекты справочников, не поддерживающих механизм подписей

        Reference.LoadSignatures(referenceObjectList1.Concat(referenceObjectList2)); // Загружаем подписи
    }
}

 

Проверка уникальности объектов

Reference.IsUniqueObjects("(Список объектов)", "(Перезагрузка объектов)") - проверяет, являются ли объекты справочника уникальными. Возвращает результаты проверки уникальности объектов

Пример
Копировать 
using System;
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using System.Linq;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        var reference = Context.Connection.ReferenceCatalog.Find(Guid.Empty).CreateReference(); // Создаем экземпляр справочника

        // Создаем объекты справочника с заполнением уникального параметра
        int referenceObjectCount = 10;
        var referenceObjects = new List<ReferenceObject>(referenceObjectCount);

        for (int i = 0; i < referenceObjectCount; i++)
        {
            ReferenceObject newObject = reference.CreateReferenceObject();
            newObject[Guid.Empty].Value = i;
            referenceObjects.Add(newObject);
        }

        // Проверяем уникальность созданных объектов. Метод возвращает результаты проверки уникальности объектов
        List<UniqueObjectCheckResult> uniqueObjectCheckResults = Reference.IsUniqueObjects(referenceObjects);

        var notUniqueObjects = uniqueObjectCheckResults.Select(cr => cr.Object).ToList(); // Получаем неуникальные объекты

        var uniqueObjects = referenceObjects.Except(notUniqueObjects); // Исключаем неуникальные объекты

        foreach (var notUniqueObject in notUniqueObjects) // Отменяем изменения для неуникальных объектов
            notUniqueObject.CancelChanges();

        Reference.EndChanges(uniqueObjects); // Фиксируем окончание изменения уникальных объектов
    }
}

 

Создание подключений

Reference.CreateComplexHierarchyLinks("(Родительские объекты)", "(Дочерние объекты)") - создает список подключений между указанными родительскими и дочерними объектами

Reference.CreateComplexHierarchyLinks("(Родительские объекты)", "(Дочерние объекты)", "(Подключение от родительских объектов)") - создает список подключений между указанными родительскими и дочерними объектами с учетом направления подключения.
Если linkFromParents true, то подключение осуществляется со стороны родительских объектов, они же являются текущими объектами в макросах-обработчиках.

Пример
Копировать 
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Users;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Для примера подключим выбранных пользователей к нескольким группам в справочнике "Группы и пользователи"
        if (Context.Reference is not UserReference userReference) // Если текущий справочник не "Группы и пользователи", то выходим
            return;

        var selectedUsers = Context.GetSelectedObjects().OfType<User>(); // Получаем выбранных пользователей

        // Находим группы пользователей
        var group1 = userReference.FindOne(userReference.ParameterGroup[UserReferenceObject.Fields.FullName], "Группа 1");
        var group2 = userReference.FindOne(userReference.ParameterGroup[UserReferenceObject.Fields.FullName], "Группа 2");

        // Создаем подключения между пользователями и группами, подключения осуществляются со стороны пользователей (т.е. будут отрабатывать обработчики событий типа "Пользователь")
        IEnumerable<ComplexHierarchyLink> complexHierarchyLinks = Reference.CreateComplexHierarchyLinks(new[] { group1, group2 }, selectedUsers, false);

        // Сохраняем подключения
        Reference.EndChanges(complexHierarchyLinks);
    }
}

 

Фиксирование окончания изменений (завершение редактирования) объектов и подключений

Reference.EndChanges("(Список объектов)") - завершает редактирование списка объектов

Reference.EndChanges("(Список подключений)") - завершает редактирование списка подключений

Пример
Копировать 
using System;
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // В коде необходимо пустые значения глобальных идентификаторов (Guid.Empty) заменить на реальные

        Reference reference = Context.Connection.ReferenceCatalog.Find(Guid.Empty).CreateReference(); // Создаем экземпляр справочника

        using var filter = new Filter(reference.ParameterGroup); // Создаем фильтр для поиска объектов в справочнике
        filter.Terms.AddTerm(reference.ParameterGroup[SystemParameterType.ObjectId], ComparisonOperator.GreaterThan, 5);
        List<ReferenceObject> referenceObjects = reference.Find(filter); // Находим объекты

        var editableObjects = new List<ReferenceObject>(referenceObjects.Count * 2); // Создаем список, в котором будем хранить редактируемые объекты

        foreach (var referenceObject in referenceObjects)
        {
            referenceObject.BeginChanges(false); // Переводим объект в состояние редактирования. Т.к. объект только что загружен, то передаем в метод false, тем самым не перезагружаем объект
            referenceObject[Guid.Empty].Value = 1; // Редактируем объект
            editableObjects.Add(referenceObject); // Добавляем в список

            // Пакетное фиксирование окончания изменений позволяет сохранять объекты разных справочников

            var linkedObject = referenceObject.GetObject(Guid.Empty); // Получаем объект по связи к 1
            if (linkedObject is not null)
            {
                linkedObject.BeginChanges(false); // Переводим в состояние редактирования объект по связи
                linkedObject[Guid.Empty].Value = "Test"; // Редактируем объект по связи
                editableObjects.Add(linkedObject); // Добавляем в список
            }
        }

        Reference.EndChanges(editableObjects); // Фиксируем окончание изменения объектов
    }
}

 

Удаление объектов и подключений

Reference.Delete("(Список объектов)") - удаляет список объектов

Reference.Delete("(Список подключений)") - удаляет список подключений

Reference.DeleteDesktopObjects("(Список Объектов)") - удаляет список объектов с учётом объектов справочника с поддержкой рабочего стола

Пример
Копировать 
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        using var filter = new Filter(Context.Reference.ParameterGroup); // Создаем фильтр для поиска объектов в текущем справочнике
        filter.Terms.AddTerm(Context.Reference.ParameterGroup[SystemParameterType.ObjectId], ComparisonOperator.GreaterThan, 5);
        List<ReferenceObject> referenceObjects = Context.Reference.Find(filter); // Находим объекты

        Reference.Delete(referenceObjects); // Удаляем объекты

        Context.RefreshReferenceWindow(); // Обновляем окно справочника
    }
}
