Связи
Команды для работы со связями.

Объект.СвязанныйОбъект ["(имя связи 1:1 или N:1)"] - возвращает связанный объект.

C#
Копировать 
Объект файл = ТекущийОбъект.СвязанныйОбъект["Файл"];
string наименованиеФайла = файл == null ? "Нет связанного файла" : файл.Параметр["Наименование"];

 

Объект.СвязанныйОбъект ["(имя связи 1:1 или N:1)"] = (объект) - устанавливает связанный объект

C#
Копировать 
Объект файл = НайтиОбъект("Файлы", "Наименование", "Мой документ.pdf");
ТекущийОбъект.СвязанныйОбъект["Файл"] = файл; // Подключает объект по связи
ТекущийОбъект.СвязанныйОбъект["Документ"] = null; // Отключает объект по связи

 

Объект.СвязанныеОбъекты ["(имя или Guid связи 1:N или N:N, или на любой справочник)"] - возвращает список связанных объектов.

C#
Копировать 
Объекты документы = ТекущийОбъект.СвязанныеОбъекты["Документы"];

foreach (Объект документ in документы)
{
    ТекущийОбъект.Отключить("Документы", документ);
}

 

Объект.Подключить("Наименование связи", (объект)) - подключает объект по связи.

C#
Копировать 
Объект файл = НайтиОбъект("Файлы", "Наименование", "Мой документ.pdf");

if (файл is not null)
    ТекущийОбъект.Подключить("Файлы", файл);

 

Объект.Отключить("Наименование связи") - отключает объект по связи 1:1, N:1
Объект.Отключить("Наименование связи", (объект)) - отключает объект по связи 1:N, N:N, или на любой справочник

C#
Копировать 
Объекты файлы = ТекущийОбъект.СвязанныеОбъекты["Файлы"];

foreach (Объект файл in файлы)
{
    if (файл.Параметр["Наименование"] == "Мой документ.pdf")
    {
         ТекущийОбъект.Отключить("Файлы ", файл);
         break;
    }
}

 

Обратные связи

Получение объектов по обратным связям

C#
Копировать 
Объект объект = ТекущийОбъект.СвязанныйОбъект["[~Один со списком]"];
Объекты объекты = ТекущийОбъект.СвязанныеОбъекты["[~Список со списком]"];

 

ИзмененнаяСвязь - возвращает наименование связи при срабатывании события "Изменение связи объекта".

C#
Копировать 
if (ИзмененнаяСвязь == "Деталь")
    ТекущийОбъект.Параметр["Наименование операции"] = "Обработка детали " + ТекущийОбъект.Параметр["Деталь->Наименование"];

 

Объект.Владелец - возвращает владельца объекта.

C#
Копировать 
Объект владелец = ТекущийОбъект.Владелец;
