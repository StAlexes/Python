Настройки конфигурации

Для дополнительной фильтрации иерархической структуры объектов справочника (например, по применяемости в изделиях, дате, контексте проектирования) выполняется настройка конфигурации, которая доступна для любого справочника системы

using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.References.StructureTypes;

public class Macro(MacroContext context) : MacroProvider(context)
{
    // Пример получения и создания настроек конфигурации
    public void WorkWithConfigurationSettings()
    {
        // Получаем единые настройки конфигурации
        ConfigurationSettings connectionSettings = Context.Connection.ConfigurationSettings;

        // Получаем особую конфигурацию справочника
        ConfigurationSettings specialReferenceSettings = Context.Reference.SpecialConfigurationSettings;

        // Создаем пустые настройки
        ConfigurationSettings emptySettings = new ConfigurationSettings(Context.Connection);

        // Копируем существующие настройки
        ConfigurationSettings clonedSettings = new ConfigurationSettings(connectionSettings);

        // Осуществляем поиск типа структуры "Конструкторская" по короткому имени "К"
        StructureTypesReferenceObject structureType = Context.Connection.References.StructureTypes.Find("К");

        // Создаем настройки с указанным активным типом структуры
        ConfigurationSettings settingsWithDesignStructure = new ConfigurationSettings(Context.Connection, structureType);
    }
}
