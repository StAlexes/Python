Соединение с сервером приложений

Для работы с данными T-FLEX DOCs, хранящимися на сервере, необходимо осуществить подключение к серверу. За соединение с сервером отвечает класс ServerConnection. В макросах соединение с сервером можно получить из контекста выполнения макроса (пример «Работа с соединением с сервером из макроса»). Во внешних приложениях для создания подключения необходимо воспользоваться статическим методом ServerConnection.Open() (пример «Соединение с сервером из внешнего приложения»).

Работа с соединением с сервером из макроса
C#
Копировать 
using System.Collections.Generic;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.Model.Signatures;
using TFlex.DOCs.Model.Stages;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        ServerConnection serverConnection = Context.Connection; // Получаем соединение с сервером из контекста макроса
        User currentUser = serverConnection.ClientView.GetUser(); // Получаем текущего пользователя

        string workingFolder = serverConnection.WorkingFolder; // Получаем путь к рабочей папке
        serverConnection.SetWorkingFolder("path to folder"); // Устанавливаем путь к рабочей папке

        ReferenceCatalog referenceCatalog = serverConnection.ReferenceCatalog; // Получаем каталог справочников
        ReferenceInfo referenceInfo = referenceCatalog.Find("Макросы"); // Находим в каталоге описание справочника по наименованию

        List<Stage> stages = serverConnection.Stages; // Получаем список всех стадий

        List<SignatureType> signatureTypes = serverConnection.SignatureTypes; // Получаем список всех типов подписей

        string serverName = serverConnection.ServerName; // Получаем имя сервера
    }
}

Соединение с сервером из внешнего приложения

В данном примере рассмотрено подключение к серверу с последующим импортом папки с файлами.
Более подробная информация о справочнике "Файлы" содержится в разделе "Специализированные справочники"

C#
Копировать 
/* Ссылки
TFlex.DOCs.Common.dll - не копируем локально (Copy local - false)
TFlex.DOCs.Model.dll  - не копируем локально (Copy local - false)
TFlex.PdmFramework.Resolve.dll - копируем локально (Copy local - true)*/


using System;
using System.Linq;
using System.Text;
using TFlex.DOCs.Common;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.References.Files;
using TFlex.PdmFramework.Resolve;

namespace ConsoleApplicationImportTest
{
    class Program
    {
        private const string FolderPath = @"C:\FilesToImport";  // Исходная папка в файловой системе, из нее будут копироваться файлы

        [STAThread]
        static void Main(string[] args)
        {
            // AssemblyResolver необходим для поиска недостающих библиотек и их автоматической загрузки
            // Указываем путь к папке Program установленного клиента DOCs
            AssemblyResolver.Instance.AddDirectory(@"C:\Program Files (x86)\T-FLEX DOCs 18\Program"); // Путь по умолчанию к клиенту DOCs

            // Внимание! Классы из зависимых библиотек, которые подгружает AssemblyResolver, не должны находиться в методе Main
            // JIT-компилятор должен обработать метод Main до его запуска, а разрешение загрузки сборок (через AssemblyResolver)
            // происходит непосредственно во время выполнения метода Main
            MainCore();
        }

        private static void MainCore()
        {
            // Подключаемся к серверу с использованием авторизации сервера
            using var serverConnection = ServerConnection.Open("Администратор", // Логин
                String.Empty, // Пароль
                "localhost", // Сервер
                null, // Конфигурация
                CommunicationMode.GRPC, // Способ передачи данных между клиентом и сервером
                DataFormatterSettings.DefaultDataSerializerAlgorithm, // Способ сериализации данных
                DataFormatterSettings.DefaultCompressionAlgorithm); // Алгоритм сжатия данных

            if (serverConnection?.IsConnected != true)
            {
                Console.WriteLine("Подключение к серверу отсутствует");
                return;
            }

            serverConnection.ConnectionLost += ServerGatewayOnConnectionLost; // Подписываемся на событие, возникающее при потере соединения с сервером

            var fileReference = new FileReference(serverConnection); // Создаем экземпляра справочника "Файлы"
            var parentFolder = fileReference.FindByRelativePath("Служебные файлы") as FolderObject; // Находим целевую папку, в которую будем импортировать

            Console.WriteLine("Импорт данных...");
            // Вызываем функцию импорта файлов
            FolderObject importedFolder = fileReference.Import(FolderPath,
                new ImportParameters
                {
                    DestinationFolder = parentFolder, // Задаем папку назначения. Если значение равно null, то импортируемая папка будет добавлена в корень справочника
                    Recursive = true, // Выполняем операцию рекурсивно для всех входящих папок
                    CreateClasses = true, // Создаем новые типы в случае их отсутствия
                    AutoCheckIn = true // Автоматическое применение изменений
                });

            WriteFilePathsFromFolder(importedFolder); // Выводим результат операции
            Console.WriteLine("Импорт данных успешно завершен");
        }

        private static void ServerGatewayOnConnectionLost(object sender, ConnectionLostEventArgs connectionLostEventArgs)
        {
            Console.WriteLine("Внимание: потеря соединения с сервером");
        }

        // Выводим все файлы папки
        private static void WriteFilePathsFromFolder(FolderObject folder)
        {
            var files = folder.Children.RecursiveLoad().OfType<FileObject>().ToList(); // Рекурсивно загружаем содержимое папки и выбираем только файлы
            var sb = new StringBuilder(files.Count);
            files.ForEach(file => sb.AppendLine(file.Path)); // Выводим относительные пути файлов
            Console.WriteLine(sb.ToString());
        }
    }
}

Результатом выполнения данного кода будет импорт папки с файлами по пути "C:\FilesToImport" (включая подпапки) в папку "Служебные файлы" справочника "Файлы".
