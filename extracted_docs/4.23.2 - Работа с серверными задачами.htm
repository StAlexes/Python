Создание серверной задачи

using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Notification.ServerTasks;
using TFlex.DOCs.Model.Notification.ServerTasks.Handlers;
using TFlex.DOCs.Model.Notification.ServerTasks.Triggers;


public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Время начала выполнения действия
        DateTime startDate = DateTime.Now.AddHours(1);

        ServerTask serverTask = new ServerTask(Context.Connection)
        {
            Name = "Тестовая задача",
            // Триггер для указания времени начала выполнения действия
            Trigger = new TimeTrigger { StartTime = startDate },
            // Выполняемое действие. В данном случае - выполнение макроса
            Action = new MacroTaskAction(Context.Connection, Guid.Empty, "Run")
        };

        // Сохраняем задачу
        serverTask.Save();
    }
}

Примечание:
Возможно использование следующих триггеров:
TimeTrigger - триггер "Однократно".
DayTrigger - триггер "Ежедневно".
WeekTrigger - триггер "Еженедельно".
MonthTrigger - триггер "Ежемесячно".

Пример удаления отключенных задач пользователя

C#
Копировать 
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Notification.ServerTasks;
using TFlex.DOCs.Model.References.Users;


public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Получаем текущего пользователя
        User currentUser = Context.Connection.ClientView.GetUser();
        // Получаем список задач
        List<ServerTask> tasks = new ServerTaskManager(Context.Connection).GetTasks();

        // Находим отключенные задачи текущего пользователя
        List<ServerTask> enabledUserTasks =
            tasks.Where(task => !task.IsDeleted && !task.Enabled && task.User == currentUser).ToList();

        if (enabledUserTasks.Count == 0)
            return;

        // Удаляем задачи
        enabledUserTasks.ForEach(userTask => userTask.Delete());
    }
}
