Изменение порядкового номера объекта

В данном примере рассмотрена обработка события "Изменение порядкового номера объекта".
Событие происходит во время изменения порядка расположения объектов при помощи команд "Переместить вверх" и "Переместить вниз"

Пример обработчика события справочника "Изменение порядкового номера объекта".
C#
Копировать 
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    {
    }

    public void ИзменениеСвязиОбъекта()
    {
        if (Context.ModelChangedArgs is not ObjectOrderChangedEventArgs orderChangedEventArgs) // Проверяем, что макрос является обработчиком события "Изменение порядкового номера объекта"
            return;

        int newOrder = orderChangedEventArgs.NewOrder; // Получаем новое значение номера
        int oldOrder = orderChangedEventArgs.OldOrder; // Получаем предыдущее значение номера
        var movingObject = orderChangedEventArgs.Sender as ReferenceObject;  // Перемещаемый объект

        bool movingDown = orderChangedEventArgs.NewOrder - orderChangedEventArgs.OldOrder > 0; // Вызвана команда "Переместить вниз"        
    }
}
