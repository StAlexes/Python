using TFlex.DOCs.Common;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Access;

public class SetAccess(MacroContext context) : MacroProvider(context)
{
    public void НазначитьДоступНаОбъектСправочника()
    {
        // Находим пользователя, для которого будет установлен доступ
        var user = Context.Connection.References.Users.FindUser("Пользователь");
        // Получаем менеджер доступа на объект. AccessRightsLoadOptions определяет режим чтения прав доступа.
        // EditorMode - в данном режиме доступы не фильтруются по направлению их действия
        var accessManager = AccessManager.GetReferenceObjectAccess(Context.ReferenceObject, AccessRightsLoadOptions.EditorMode);
        accessManager.IsInherit = false; // Установка явного доступа на объект.

        // Создание нового доступа.
        var accessGroup = new AccessGroup(Context.Connection)
        {
            Type = AccessType.Object,
            Name = "Новый доступ"
        };

        // Устанавливаем состояния операций доступа
        accessGroup.SetCommandState(AccessType.Object.Read, AccessCommandState.Allowed); // Чтение разрешено.
        accessGroup.SetCommandState(AccessType.Object.Change, AccessCommandState.Allowed); // Редактирование разрешено. 
        accessGroup.SetCommandState(AccessType.Object.Print, AccessCommandState.Forbidden); // Печать запрещена.

        accessManager.SetAccess(user, accessGroup); // Установка доступа для пользователя.
        accessManager.Save();
    }
}
