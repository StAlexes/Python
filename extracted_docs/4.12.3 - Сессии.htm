Сессии
Пример по работе с журналом событий "Сессии"

В данном примере рассмотрена работа с журналом событий "Сессии".
Журнал содержит информацию о таких событиях, как успешный вход пользователя в систему, выход из системы и др.

using System;
using System.Collections.Generic;
using System.Globalization;
using TFlex.DOCs.Data.Logging.SessionsLog;
using TFlex.DOCs.Model.Logging;
using TFlex.DOCs.Model.Logging.Search;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Search;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Менеджер журнала 'Сессии'
        var sessionsLogManager = new SessionsLogManager(Context.Connection);
        var sessionLogEvents = sessionsLogManager.Load(CreateSessionLogFilter()); // Загружаем события

        List<long> eventIds = new List<long>(sessionLogEvents.Count);
        foreach (SessionsLog sessionLogEvent in sessionLogEvents)
        {
            var eventId = sessionLogEvent.EventId; // Идентификатор события
            var eventEventTime = sessionLogEvent.EventTime; // Время события (UTC)
            var configuration = sessionLogEvent.Configuration; // Конфигурация работы
            var eventType = sessionLogEvent.EventType; // Тип события

            eventIds.Add(eventId);
        }

        // Получаем детали событий журнала 'Сессии'
        var sessionsDetails = sessionsLogManager.GetSessionLogEventDetailObjects(eventIds.ToArray());
        foreach (var sessionDetails in sessionsDetails)
        {
            SessionEventBaseLogDescription details = sessionDetails.Value;
            switch (details) // В зависимости от типа получаем необходимые данные из детальной информации по событию
            {
                case SessionEventLoginFailedLogDescription sessionEventLoginFailedLogDescription: // Описание события, связанного с ошибкой авторизации
                    string clientVersion = sessionEventLoginFailedLogDescription.ClientVersion; // Версия клиента
                    bool windowsAuthentication = sessionEventLoginFailedLogDescription.WindowsAuthentication; // Признак аутентификации Windows
                    break;
                default:
                    continue;
            }
        }
    }

    private static Filter CreateSessionLogFilter()
    {
        var filter = new Filter();

        // Фильтруем события по типу
        var eventTypes = new[] { SessionLogEventType.LoginSuccessful, SessionLogEventType.Logout, SessionLogEventType.LoginFailed };
        filter.Terms.Add(new LogTerm(SessionsLogFields.EventType, ComparisonOperator.IsOneOf, eventTypes));

        // Получаем события за последний месяц
        filter.Terms.Add(new LogTerm(SessionsLogFields.EventTime,
            ComparisonOperator.GreaterThanOrEqual, DateTime.Now.AddMonths(-1).ToString(CultureInfo.InvariantCulture)));

        filter.Terms.Add(new LogTerm(SessionsLogFields.UserId, ComparisonOperator.Equal, 2)); // Получаем события по конкретному пользователю

        return filter;
    }
}
