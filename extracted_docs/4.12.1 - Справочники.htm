Справочники
Пример по работе с журналом событий "Справочники"

В данном примере рассмотрена работа с журналом событий "Справочники".
Данный журнал содержит информацию о событиях, таких как создание, изменение, удаление объектов и подключений, и др.
Для получения событий по конкретному справочнику необходимо, чтобы в свойствах данного справочника был активирован флаг "Поддержка журнала"

using System;
using System.Collections.Generic;
using System.Globalization;
using TFlex.DOCs.Data.Logging.DataLog;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Logging;
using TFlex.DOCs.Model.Logging.Search;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Search;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        // Менеджер журнала изменения данных справочников
        var dataLogManager = new DataLogManager(Context.Connection);

        // Получить информацию о всех справочниках, для которых включено журналирование, можно следующим образом:
        // var referencesWithDataLogging = dataLogManager.GetReferencesWithDataLogging();

        // Получаем описание справочника, для которого необходимо получить события изменения данных
        ReferenceInfo referenceInfo = Context.Connection.ReferenceCatalog.Find(Guid.Empty);
        Filter filter = CreateDataLogFilter(); // Создаем фильтр для событий        
        List<DataLog> dataChangeEvents = dataLogManager.Load(referenceInfo.Id, filter); // Загружаем события изменения данных
        List<long> eventIds = new List<long>(dataChangeEvents.Count);

        foreach (DataLog dataChangeEvent in dataChangeEvents) // Обрабатываем полученные события
        {
            long eventId = dataChangeEvent.EventId; // Идентификатор события
            DateTime eventEventTime = dataChangeEvent.EventTime; // Время события (UTC)
            string user = dataChangeEvent.User; // Пользователь, выполнивший действие
            string host = dataChangeEvent.Host; // Рабочее место, на котором было выполнено действие
            DataLogEventType eventType = dataChangeEvent.EventType; // Тип события

            eventIds.Add(eventId);
        }

        // Получаем детали событий журнала 'Справочники'
        var eventDetails = dataLogManager.GetDataLogEventDetailObjects(referenceInfo.Id, eventIds.ToArray());

        foreach (var eventDetail in eventDetails)
        {
            DataLogBaseLogDescription description = eventDetail.Value; // Получаем базовое детальное описание

            switch (description)
            {
                case ObjectCreateLogDescription objectCreateLogDescription: // Обрабатываем событие создания объекта
                    var parameters = objectCreateLogDescription.Parameters; // Параметры созданного объекта
                    var links = objectCreateLogDescription.Links; // Информация о подключенных объектах
                    break;
                case SignatureUpdateLogDescription signatureUpdateLogDescription: // Обрабатываем описание события обновления подписи
                    var previousSignature = signatureUpdateLogDescription.PreviousSignature; // Предыдущие свойства подписи
                    break;
                default:
                    continue;
            }
        }
    }

    private Filter CreateDataLogFilter()
    {
        var filter = new Filter();

        // Фильтруем события по типу
        DataLogEventType[] eventTypes = new[] { DataLogEventType.Create, DataLogEventType.Delete, DataLogEventType.Update };
        filter.Terms.Add(new LogTerm(DataLogFields.EventType, ComparisonOperator.IsOneOf, eventTypes));

        // Получаем события за последние 10 минут
        filter.Terms.Add(new LogTerm(DataLogFields.EventTime,
            ComparisonOperator.GreaterThanOrEqual, DateTime.Now.AddMinutes(-10).ToString(CultureInfo.InvariantCulture)));

        // Можно указать Guid-ы объектов, по которым будут получены события
        filter.Terms.Add(new LogTerm(DataLogFields.ObjectGuid, ComparisonOperator.IsOneOf, new[] { Guid.Empty }));

        // Получаем только действия, выполненные текущим пользователем
        filter.Terms.Add(new LogTerm(DataLogFields.UserId, ComparisonOperator.Equal, Context.Connection.ClientView.UserId));

        return filter;
    }
}
