Работа с параметрами типа "Иконка" и "Растровое изображение"
C#
Копировать 
using System;
using System.Drawing;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Icons;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        // Получаем изображение и иконку из справочника "Изображения"
        var imageReference = new ImageReference(Context.Connection);
        var imageObject = imageReference.Find(new Guid("Guid объекта")); // Находим объект справочника "Изображения" с типом 'Растровое изображение' 
        var iconObject = imageReference.Find(new Guid("Guid объекта")); // Находим объект справочника "Изображения" с типом 'Иконка' 

        if (imageObject is null || iconObject is null)
            return;

        Image image = null;
        if (imageObject.IsImage) // Проверяем, является ли полученный объект изображением
            image = imageObject.Image.GetImage(); // Получаем изображение из параметра

        if (iconObject.IsIcon) // Проверяем, является ли полученный объект иконкой
        {
            IconImage iconImage = iconObject.Icon.Value; // Получаем иконку T-FLEX DOCs

            if (iconImage.IsSvgImage) // Проверяем, является ли иконка векторной
            {
                byte[] svgIcon = iconImage.SvgSource;
            }
            else
            {
                Icon icon = iconImage.InternalIcon; // Получаем иконку типа Icon
                Image mediumIcon = iconImage.MediumImage; // Получаем иконку среднего размера
            }
        }

        // Сохраняем в текущий объект изображение из справочника "Изображения"
        if (Context.ReferenceObject is null || image is null)
            return;

        bool changingObject = Context.ReferenceObject.Changing;
        try
        {
            if (!changingObject)
                Context.ReferenceObject.BeginChanges(false);

            var converter = new ImageConverter(); // Получаем конвертер изображений
            // Запись иконки или изображения в параметр требует преобразования в массив байт
            Context.ReferenceObject[new Guid("Guid параметра с типом 'Растровое изображение'")].Value = (byte[])converter.ConvertTo(image, typeof(byte[]));

            if (!changingObject)
                Context.ReferenceObject.EndChanges();
        }
        catch
        {
            if (!changingObject)
                Context.ReferenceObject.CancelChanges();
        }
    }
}

Работа со строковыми параметрами, содержащими текст в формате html и rtf

/* Ссылки

using System;
using DevExpress.XtraRichEdit;
using DevExpress.XtraRichEdit.API.Native;
using TFlex.DOCs.Common;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class RtfHtmlEditor : MacroProvider
{
    private Guid _parameterGuid = new("Guid параметра"); // Глобальный идентификатор параметра, содержащего форматированный текст

    public RtfHtmlEditor(MacroContext context) : base(context)
    { }

    /// <summary>Конвертирует текст в различные форматы</summary>
    public void ConvertText()
    {
        if (Context.ReferenceObject is null) // Проверяем наличие текущего объекта
            return;

        string textValue = Context.ReferenceObject[_parameterGuid].GetString(); // Получаем значение текстового параметра
        using var reDocServer = new RichEditDocumentServer(); // Создаем класс, предоставляющий функции обработки форматированного текста

        // Задаем текст для компонента RichEditDocumentServer в зависимости от формата
        if (RtfHtmlManager.IsHtmlString(textValue)) // Проверяем, хранится ли текст в html-формате
            reDocServer.HtmlText = textValue;
        else if (RtfHtmlManager.IsRtfString(textValue)) // Проверяем, хранится ли текст в rtf-формате
            reDocServer.RtfText = textValue;
        else
            reDocServer.Text = textValue;

        // RichEditDocumentServer умеет конвертировать текст в различные форматы. Для этого достаточно обратиться к свойству, отвечающему за необходимый формат

        string html = reDocServer.HtmlText; // Получаем текст в html-формате
        string rtf = reDocServer.RtfText; // Получаем текст в rtf-формате
        string text = reDocServer.Text; // Получаем обычный текст
    }

    /// <summary>Изменяет значение параметра заданного формата</summary>
    public void ModifyParameterValue()
    {
        if (Context.ReferenceObject is null) // Проверяем наличие текущего объекта
            return;

        string textValue = Context.ReferenceObject[_parameterGuid].GetString(); // Получаем значение текстового параметра

        // Для примера считаем, что нам необходимо обработать текст в формате rtf
        if (!RtfHtmlManager.IsRtfString(textValue))
            return;

        using var reDocServer = new RichEditDocumentServer() { RtfText = textValue }; // Создаем класс, предоставляющий функции обработки форматированного текста

        reDocServer.Document.BeginUpdate(); // Начинаем редактирование
        reDocServer.Document.AppendText("\r\nНовая строка"); // Добавляем текст

        // Для добавления уже размеченного текста можно воспользоваться методами AppendHtmlText и AppendRtfText
        // Находим все вхождения заданной строки c учетом регистра. Каждое вхождение представляет собой диапазон с начальной и конечной позициями в документе
        DocumentRange[] documentRanges = reDocServer.Document.FindAll("Текст1", SearchOptions.CaseSensitive);

        if (documentRanges.Length > 0) // Заменяем первое вхождение
            reDocServer.Document.Replace(documentRanges[0], "Текст2");

        reDocServer.Document.EndUpdate(); // Завершаем редактирование

        // Обновляем значение параметра
        Context.ReferenceObject.Modify(ro => ro[_parameterGuid].Value = reDocServer.RtfText);
    }
}
