Ревизии

Ревизия - это копия объекта, имеющая логическую связь с исходным объектом. При этом различные ревизии одного и того же объекта могут использоваться независимо друг от друга. Ревизии позволяют внести изменения в объект, не затрагивая его существующую версию.
Более подробное описание ревизий приведено в разделе "Специальные возможности системы при работе с данными" документа "T-FLEX DOCs - Руководство пользователя"

Пример создания ревизии
C#
Копировать 
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Revisions;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        ReferenceObject referenceObject = Context.ReferenceObject; // Для примера создадим ревизию у текущего объекта

        if (referenceObject is null)
            return;

        if (!referenceObject.Reference.ParameterGroup.SupportsRevisions) // Определяем, поддерживает ли справочник механизм ревизий
            return;

        IList<RevisionLevelObject> revisionLevels = referenceObject.Class.RevisionNamingRule.RevisionLevels.AsList; // Получаем уровни ревизии

        if (revisionLevels.Count == 0)
            return;

        // Находим минорную ревизию (объект из списка объектов "Уровни ревизии" правила именования ревизий)
        RevisionLevelObject revisionLevel = revisionLevels
            .FirstOrDefault(rl => rl.Name.GetString().Equals("Минорная")) ?? revisionLevels.First();

        // Определяем родительский объект
        ReferenceObject parent = referenceObject.Reference.ParameterGroup.HierarchyType == TFlex.DOCs.Model.Structure.ReferenceHierarchyType.Simple
            ? referenceObject.Parent
            : null;

        ReferenceObject revision = referenceObject.CreateRevision(revisionLevel, parent, false); // Создаем ревизию

        // У созданных ревизий уже зафиксированы изменения, вызывать EndChanges() не нужно, но если включена поддержка истории изменений, то можно применить изменения на рабочем столе

        if (revision.CanCheckIn)
            revision.CheckIn("Применение изменений", false);
    }
}

Пример получения ревизий объекта и установки уникальности
C#
Копировать 
using System;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        ReferenceObject referenceObject = Context.ReferenceObject; // Для примера получим ревизии текущего объекта

        if (referenceObject is null)
            return;

        if (!referenceObject.Reference.ParameterGroup.SupportsRevisions) // Определяем, поддерживает ли справочник механизм ревизий
            return;

        // Через системные параметры можно получить информацию о ревизиях
        string revisionName = referenceObject.SystemFields.RevisionName; // Имя ревизии
        bool revisionIsActual = referenceObject.SystemFields.IsActualRevision; // Актуальность ревизии
        string lastRevisionName = referenceObject.SystemFields.LastRevisionName; // Имя последней ревизии
        string sourceRevisionName = referenceObject.SystemFields.SourceRevisionName; // Имя исходной ревизии. У исходной ревизии возвращается пустая строка

        // Получаем копию параметров загрузки и указываем загрузку без учета настроек конфигурирования
        var settings = referenceObject.Reference.LoadSettings.Clone();
        settings.UseConfigurationSettings = false;

        // Создаем фильтр для поиска ревизий объекта. Поиск осуществляется по системному параметру "Идентификатор логического объекта ревизии"
        using Filter filter = new Filter(referenceObject.Reference.ParameterGroup);
        filter.Terms.AddTerm(referenceObject.Reference.ParameterGroup.SystemParameters.Find(SystemParameterType.LogicalObjectGuid)
                    , ComparisonOperator.Equal
                    , referenceObject.SystemFields.LogicalObjectGuid);

        // Получаем ревизии при помощи фильтра
        var revisions = referenceObject.Reference.Find(filter, -1, null, loadSettings: settings);

        // Ревизии без учета параметров загрузки можно получить более простым способом:
        // var revisions = referenceObject.GetRevisions();

        // Получаем любую неисходную и актуальную ревизию
        var actualRevision = revisions.FirstOrDefault(r => r.SystemFields.IsActualRevision && !String.IsNullOrEmpty(r.SystemFields.SourceRevisionName));

        // Делаем ревизию неактуальной
        if (actualRevision is not null)
            actualRevision.SystemFields.IsActualRevision = false;
    }
}
