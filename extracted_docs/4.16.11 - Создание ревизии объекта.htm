Создание ревизии объекта

В данном примере рассмотрена обработка события "Создание ревизии объекта".
Событие происходит во время создания ревизии объекта справочника.

Пример обработчика события "Создание ревизии объекта"
C#
Копировать 
using TFlex.DOCs.Model.Macros;

public class CreationRevision(MacroContext context) : MacroProvider(context)
{
    // Обработчик события "Создание ревизии объекта"
    public override void Run()
    {
        var newRevision = Context.ReferenceObject; // В данном обработчике текущим объектом является созданная ревизия
        string newRevisionName = newRevision.SystemFields.RevisionName; // Получаем имя новой ревизии

        var oldRevision = newRevision.Prototype; // Получаем исходный объект/ревизию, на основе которого создается ревизия
        string oldRevisionName = oldRevision.SystemFields.RevisionName; // Получаем наименование ревизии исходного объекта
    }
}
