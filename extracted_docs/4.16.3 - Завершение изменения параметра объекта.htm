Завершение изменения параметра объекта

В данном примере рассмотрена обработка события "Завершение изменения параметра объекта", возникающего после изменения значения параметра

C#
Копировать 
using System;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class ChangedParameterEventHandler : MacroProvider
{
    private Guid _fullName = new Guid("bf5febb9-3478-4a78-b5d6-cc3101c4af04"); // "ФИО"
    private Guid _nameGuid = new Guid("581ce731-39c3-4fb4-ae3f-6a6ecbf30185"); // "Имя"
    private Guid _patronymicGuid = new Guid("b3071029-5c5b-459e-8125-851fcb729476"); // "Фамилия"
    private Guid _surnameGuid = new Guid("1aa4aa20-a9f7-4760-b46c-a72e0b9f3dab"); // "Отчество"

    public ChangedParameterEventHandler(MacroContext context)
        : base(context)
    { }

    // Обработчик события "Завершение изменения параметра объекта"
    public override void Run()
    {
        if (Context.ModelChangedArgs is not ObjectParameterChangedEventArgs args) // Проверяем, является ли макрос обработчиком события "Завершение изменения параметра объекта"
            return;

        var fullNameParameters = new[] { _nameGuid, _patronymicGuid, _surnameGuid };
        if (!fullNameParameters.Contains(args.Parameter.ParameterInfo.Guid)) // Если изменяется параметр, не относящийся к ФИО, выходим
            return;

        string name = Context.ReferenceObject[_nameGuid].GetString();
        string patronymic = Context.ReferenceObject[_patronymicGuid].GetString();
        string surname = Context.ReferenceObject[_surnameGuid].GetString();

        var previousValue = args.OldValue; // Если необходимо получить предыдущее значение

        Context.ReferenceObject[_fullName].Value = String.Format("{0} {1} {2}", name, patronymic, surname);
    }
}
