В данном примере показан обработчик события сервера

Пример обработчика события сервера
C#
Копировать 
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Notification.ServerEventHandlers;
using TFlex.DOCs.Model.References;

public class ServerEvent : MacroProvider
{
    public ServerEvent(MacroContext context)
        : base(context)
    { }

    // Обработчик события сервера
    public override void Run()
    {
        if (Context.ModelChangedArgs is not ServerEventArgs serverEventArgs) // Получаем аргументы события сервера
            return;

        ServerEventHandler serverEventHandler = serverEventArgs.ServerEventHandler; // Получаем обработчик события сервера
        var serverEventGuid = serverEventHandler.Guid; // Получаем Guid собятия сервера

        EventType eventType = serverEventArgs.ServerEventHandler.EventType; // Получаем тип события сервера

        bool validEventType = (EventType.Created | EventType.Changed).HasFlag(serverEventArgs.ServerEventHandler.EventType); // Проверяем, обрабатывается ли событие типа "Создание" и/или "Изменение"
    }
}
