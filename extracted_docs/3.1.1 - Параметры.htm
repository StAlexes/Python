Параметры
Команды для работы с параметрами

Объект.Параметр ["(имя или Guid параметра)"] - возвращает значение параметра указанного объекта.

C#
Копировать 
string наименование = договор.Параметр["Наименование"];
string имя = пользователь.Параметр["d6e2c40c-e3fe-401d-bf48-c9980992c47e"]; // Получаем значение параметра по его идентификатору
  Внимание

Если наименование параметра совпадает с системным («Guid», «Имя» и т.д.), то для получения или установки значения такого параметра необходимо использовать его глобальный уникальный идентификатор (Guid)

Параметр ["(имя или Guid параметра)"] - возвращает значение параметра текущего объекта.

C#
Копировать 
string но = Параметр["Наименование"] + " (" + Параметр["Обозначение"] + ")";

 

Объект.Параметр ["(имя связи 1:1 или N:1->имя параметра)"] - возвращает значение параметра связанного объекта.

C#
Копировать 
string наименованиеФайла = договор.Параметр["Файл->Наименование"];

 

Параметр ["(имя связи 1:1 или N:1->имя параметра)"] - возвращает значение параметра объекта, связанного с текущим.

C#
Копировать 
string наименованиеФайла = Параметр["Файл->Наименование"];

 

Объект.Параметр ["(имя параметра)"] = (значение) - устанавливает значение параметра указанного объекта.

C#
Копировать 
договор.Параметр["Срок годности"] = DateTime.Now.AddYears(1);

 

Параметр ["(имя или Guid параметра)"] = (значение) - устанавливает значение параметра текущего объекта.

C#
Копировать 
Параметр["Срок годности"] = DateTime.Now.AddYears(1);

 

Тип.Параметры - возвращает список имен параметров указанного типа.

C#
Копировать 
Объект1.Изменить();

foreach (string параметр in Объект1.Тип.Параметры)
{
    Объект1.Параметр[параметр] = Объект2.Параметр[параметр];
}

Объект1.Сохранить();

 

ГлобальныйПараметр ["имя или Guid глобального параметра"] - возвращает значение глобального параметра

C#
Копировать 
ТекущийОбъект.Параметр["Номер договора"] = ГлобальныйПараметр["Номер следующего договора"];

 

ГлобальныйПараметр ["имя или Guid глобального параметра"] = (значение) - устанавливает значение глобального параметра

C#
Копировать 
ГлобальныйПараметр["Номер следующего договора"] = ГлобальныйПараметр["Номер следующего договора"] + 1;

 

ИзмененныйПараметр - если макрос является обработчиком события «Изменение параметра объекта» или «Завершение изменения параметра объекта», то возвращает название измененного параметра

C#
Копировать 
if (ИзмененныйПараметр == "Длина стороны")
    ТекущийОбъект.Параметр["Площадь"] = Math.Pow(ТекущийОбъект.Параметр["Длина стороны"], 2);

 

ПолучитьСписокЗначений("Параметр", "Справочник"[, "Список объектов"]) - возвращает список значений для параметра справочника или списка объектов.

C#
Копировать 
СписокЗначений списокЗначений = ПолучитьСписокЗначений("Важность", "Поручения"); // Получаем список значений для параметра "Важность"

foreach (var значениеСписка in списокЗначений)
{
    string наименование = значениеСписка.Наименование; // Получаем наименование значения параметра ("Низкая", "Средняя", "Высокая")
    var значение = значениеСписка.Значение; // Получаем соответствующее значение параметра (0, 1, 2)
}

 

Пример обработчика события справочника «Завершение изменения параметра объекта» с помощью макроязыка.
C#
Копировать 
using System;
using System.Linq;
using TFlex.DOCs.Model.Macros;

public class Macro(MacroContext context) : MacroProvider(context)
{
    // Обработчик события "Завершение изменения параметра объекта"
    public override void Run()
    {
        // Определяем наименования параметров
        const string параметрФИО = "ФИО";
        const string параметрИмя = "Имя";
        const string параметрФамилия = "Фамилия";
        const string параметрОтчество = "Отчество";
        string[] параметрыПолногоИмени = [параметрИмя, параметрФамилия, параметрОтчество];

        // Если параметр не изменялся или изменен параметр, не относящийся к составу полного имени, выходим
        if (String.IsNullOrEmpty(ИзмененныйПараметр) || !параметрыПолногоИмени.Contains(ИзмененныйПараметр))
            return;

        // Получаем значения параметров
        string имя = Параметр[параметрИмя];
        string фамилия = Параметр[параметрФамилия];
        string отчество = Параметр[параметрОтчество];

        Параметр[параметрФИО] = $"{фамилия} {имя} {отчество}".Trim(); // Записываем полное имя в параметр
    }
}
