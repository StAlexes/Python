Руководство по T-FLEX DOCs Open API

Импорт данных из Microsoft Project

API T-FLEX DOCs позволяет импортировать проекты из MS Project 2010 в DOCs. Ниже приводится пример реализации такой возможности.

Создание структуры для обмена данными

Для того чтобы производить импорт данных, следует создать структуру для обмена данными. Реализуем ее в виде класса MSImportExchangeFormat:

C#
Копировать 
// Структура для обмена данными
internal class MSImportExchangeFormat
{
    ...
};

Теперь будем добавлять переменные, методы и проч. в класс MSImportExchangeFormat.
В первую очередь объявим экземпляр приложения MS Project и наименование для импортируемого проекта:

C#
Копировать 
// Экземпляр приложения
public MSProject.Application app = null;
// Наименование проекта
internal string ProjectName { get; set; }

Затем займемся подпроектами:

C#
Копировать 
// Подпроект
internal class MSImportSubProject
{
    // GUID подпроекта
    internal Guid GUID { get; set; }
    // Наименование
    internal string Name { get; set; }
    // Подпроект в DOCs
    internal BaseTaskObject AssignedTask { get; set; }
    // Конструктор
    public MSImportSubProject(Guid guid, string name)
    {
        // GUID подпроекта
        GUID = guid;
        // Наименование
        Name = name;
    }
}
// Список подпроектов
internal ArrayList _subprojects = new ArrayList();
// Добавление нового подпроекта
internal int AddSubProject(Guid guid, string name)
{
    _subprojects.Add(new MSImportSubProject(guid, name));
    return 0;
}

И, наконец, задания:

C#
Копировать 
// Задания
internal class MSImportTask
{
    // GUID задания
    internal Guid Guid { get; set; }
    // Родительский объект
    internal Guid ParentGuid { get; set; }
    // Предшествующие задания
    internal ArrayList PredecessorTasksGuids { get; set; }
    // Наименование
    internal string Name { get; set; }
    // Описание
    internal string Description { get; set; }
    // Дата начала
    internal DateTime StartDate { get; set; }
    // Дата окончания
    internal DateTime FinishDate { get; set; }
    // Прогресс выполнения
    internal int CompletePercent { get; set; }
    // Задание в DOCs
    internal BaseTaskObject AssignedTask { get; set; }
    // Конструктор
    public MSImportTask(Guid guid, Guid parent, ArrayList predecessorTasks, string name, string description, DateTime startDate, DateTime finishDate, int completePercent)
    {
        Guid = guid;
        ParentGuid = parent;
        PredecessorTasksGuids = predecessorTasks;
        Name = name;
        Description = description;
        StartDate = startDate;
        FinishDate = finishDate;
        CompletePercent = completePercent;
    }
}
// Список заданий
internal ArrayList _tasks = new ArrayList();
// Добавление нового задания
internal int AddTask(Guid guid, Guid parent, ArrayList predecessorTasks, string name, string description, DateTime startDate, DateTime finishDate, int completePercent)
{
    _tasks.Add(new MSImportTask(guid, parent, predecessorTasks, name, description, startDate, finishDate, completePercent));
    return 0;
}

Кроме того, нам также понадобится специальная функция для нахождения соответствий объектов в DOCs и MS Project:
(PS. Дело в том, что каждый импортируемый объект в MS Project имеет определенный GUID, а при создании нового соответствующего объекта в DOCs для него генерируется новый GUID.)

C#
Копировать 
// Поиск объекта в DOCs по GUID данного объекта в MSProject
internal BaseTaskObject FindAssignedObject(Guid guid)
{
    // Поиск в подпроектах
    foreach (MSImportSubProject subproject in _subprojects)
    {
        if (subproject.GUID == guid)
            return subproject.AssignedTask;
    }

    // Поиск в заданиях
    foreach (MSImportTask task in _tasks)
    {
        if (task.Guid == guid)
            return task.AssignedTask;
    }

    // Возврат с кодом неуспешно
    return null;
}

Будет также весьма полезна функция форматированного вывода всего содержимого экземпляра класса MSImportExchangeFormat:

C#
Копировать 
// Печать содержимого
public void PrintContent()
{
    // Формирование содержимого
    string data = "";

    foreach (MSImportTask task in _tasks)
    {
        data += "*** Задание \n";
        data += "GUID задания:\t" + task.Guid.ToString() + "\n";
        data += "Родительский:\t" + task.ParentGuid.ToString() + "\n";

        foreach (Object predecessorTask in task.PredecessorTasksGuids)
        {
            data += "Предшествующий:\t" + predecessorTask.ToString() + "\n";
        }

        data += "Наименование:\t" + task.Name + "\n";
        data += "Описание работы:\t" + task.Description + "\n";
        data += "Начало работы:\t" + task.StartDate.ToShortDateString() + "\n";
        data += "Конец работы:\t" + task.FinishDate.ToShortDateString() + "\n";
        data += "Прогресс, %:\t" + task.CompletePercent.ToString() + "\n";
    }

    // Вывод содержимого
    if (data == "")
    {
        Log("(Данные отсутствуют)");
    }
    else
    {
        Log(data);
    }
}

Напоследок организуем в классе возможность вывода лога:

C#
Копировать 
public delegate void MsgEventHandler(object sender, MsgEventArgs e);

// Событие Msg - вывод лога
public event MsgEventHandler Msg;

protected virtual void OnMsg(object sender, MsgEventArgs e)
{
    Msg(this, e);
}

// Формирование лога
public void Log(string text)
{
    if (text == "")
    {
        Msg(this, new MsgEventArgs(text, true));
    }
    else
    {
        Msg(this, new MsgEventArgs(text, false));
    }
}

// Аргументы и делегат Msg
public class MsgEventArgs : EventArgs
{
    internal string Text { get; set; }
    internal bool Clear { get; set; }
    public MsgEventArgs(string text, bool clear)
    {
        Text = text;
        Clear = clear;
    }
}
Извлечение данных

После того как создана структура для обмена данными, можно приступить к созданию функции для извлечения данных из файла MS Project:

C#
Копировать 
// Извлечение данных из файла
public int ExtractData(string fileName)
{
    // Проверка входных данных
    if (!File.Exists(fileName))
    {
        Log("Невозможно открыть файл " + fileName);
        return -1;
    }

    // Открытие и обработка файла
    try
    {
        // Запуск приложения
        app = new MSProject.Application();
        app.Visible = false;
        // Открытие файла
        app.FileOpen(fileName, true, Type.Missing, Type.Missing, Type.Missing, Type.Missing, Type.Missing, Type.Missing, Type.Missing, Type.Missing, Type.Missing, MSProject.PjPoolOpen.pjPoolReadOnly, Type.Missing, Type.Missing, Type.Missing, Type.Missing);
        // Чтение данных
        Log("Чтение данных..");
        ProjectName = app.Name;

        foreach (MSProject.Project proj in app.Projects)
        {
            AddSubProject(new Guid(proj.GetServerProjectGuid()), proj.Name);]

            foreach (MSProject.Task task in proj.Tasks)
            {
                ArrayList al = new ArrayList();

                foreach (MSProject.Task predecessorTask in task.PredecessorTasks)
                {
                    al.Add(new Guid(predecessorTask.Guid));
                }

                AddTask(new Guid(task.Guid), new Guid(task.OutlineParent.Guid), al, task.Name, task.Notes, Convert.ToDateTime(task.ScheduledStart), Convert.ToDateTime(task.ScheduledFinish), Convert.ToInt32(task.PercentComplete));
            }
        }
    }
    catch (Exception ex)
    {
        // Вывод сообщения об ошибке, возврат с кодом неуспешно
        Log(ex.Message + ": " + ex.InnerException.Message);
        return -1;
    }

    finally
    {
        // Закрытие приложения
        if (app != null)
        {
            app.Quit(MSProject.PjSaveType.pjDoNotSave);
        }
    }
    // Возврат с кодом успешно
    Log("Чтение данных успешно завершено");
    return 0;
}
Импорт данных

Извлеченные данные можно импортировать следующим образом:

C#
Копировать 
// Импорт данных
public int ImportData(Guid parentProjectFolderGUID)
{
    // Проверка входных данных
    if (ProjectName == null)
    {
        Log("Невозможно импортировать данные");
        return -1;
    }

    Log("Импорт данных");
    // Подготовка данных
    ReferenceInfo referenceInfo = ReferenceCatalog.FindReference(new Guid("9a7daed1-7c2c-4f16-8147-6423d8eea413"));
    Reference reference = referenceInfo.CreateReference();
    ReferenceObject projectParentObject = reference.Find(parentProjectFolderGUID);
    // Импорт данных
    try
    {
        Log("Создание подпроектов..");

        foreach (MSImportExchangeFormat.MSImportSubProject subproject in _subprojects)
        {
            BaseTaskObject _subproject = reference.CreateReferenceObject(projectParentObject,
            reference.Classes.Find(TFlex.DOCs.Model.References.Projects.ProjectTypes.Keys.Project)) as BaseTaskObject;
            _subproject.Name = subproject.Name;
            _subproject.EndChanges();
            // Создание работ
            Log("Создание работ..");
            ReferenceObjectSaveSet taskSet = new ReferenceObjectSaveSet();

            foreach (MSImportExchangeFormat.MSImportTask task in _tasks)
            {
                DateTime start = DateTime.Now;
                BaseTaskObject _task = reference.CreateReferenceObject(_subproject,
                reference.Classes.Find(TFlex.DOCs.Model.References.Projects.ProjectTypes.Keys.Task)) as BaseTaskObject;
                taskSet.Add(_task);
                _task.Name = task.Name;
                _task.Description = task.Description;
                _task.StartTime = task.StartDate;
                _task.EndTime = task.FinishDate;
                _task.Progress = Convert.ToDouble(task.CompletePercent) / 100;
                //_task.EndChanges();
                task.AssignedTask = _task;
            }
            taskSet.EndChanges();
            // Установка связей для работ
            Log("Установка связей для работ..");

            foreach (MSImportExchangeFormat.MSImportTask task in _tasks)
            {
                BaseTaskObject _task = FindAssignedObject(task.Guid);
                _task.BeginChanges();

                if (FindAssignedObject(task.ParentGuid) != null)
                {
                    _task.SetParent(FindAssignedObject(task.ParentGuid));
                }

                _task.EndChanges();

                foreach (Guid predessor in task.PredecessorTasksGuids)
                {
                    (FindAssignedObject(task.Guid) as TaskObject).AddNewLink(FindAssignedObject(predessor) as TaskObject, TaskLinkType.StartToFinish, 0);
                }
            }
        }
    }
    catch (Exception ex)
    {
        // Вывод сообщения об ошибке, возврат с кодом неуспешно
        Log(ex.Message + ": " + ex.InnerException.Message);
        return -1;
    }
    // Возврат с кодом успешно
    Log("Импорт данных успешно завершен");
    return 0;
}
Использование с визуальными компонентами

Описанный выше класс для импорта данных может быть использован с визуальными компонентами следующим образом.

Объявление и инициализация класса MSImportExchangeFormat:

C#
Копировать 
// Создание объекта MSImportExchangeFormat для импорта данных
MSImportExchangeFormat msief = new MSImportExchangeFormat();

Организация ведения лога (где richTextBoxLog - поле для вывода лога):

C#
Копировать 
// Добавление обработчика событий MSImportExchangeFormat
msief.Msg += new MsgEventHandler(msief_Msg);

// Вывод лога объекта MSImportExchangeFormat
public void msief_Msg(object sender, MsgEventArgs e)
{
    if (e.Clear == true)
        richTextBoxLog.Clear();

    richTextBoxLog.AppendText(e.Text + '\n');
}

Выбор целевой папки для импортируемых данных (где textBoxTargetFolder - наименование/идентификатор целевой папки DOCs для импортируемых данных):

C#
Копировать 
// Диалог выбора папки в DOCs для импортируемых данных
ReferenceInfo referenceInfo = ReferenceCatalog.FindReference(new Guid("9a7daed1-7c2c-4f16-8147-6423d8eea413"));
Reference reference = referenceInfo.CreateReference();
TFlexDOCsClientUI.Initialize();

using (ISelectReferenceObjectDialog dlg = ObjectCreator.CreateObject<ISelectReferenceObjectDialog>())
{
    dlg.Initialize(referenceInfo);

    if (dlg.ShowDialog(null) != DialogOpenResult.Ok)
        return;

    textBoxTargetFolder.Tag = dlg.FocusedObject.SystemFields.Guid.ToString();
    textBoxTargetFolder.Text = (dlg.FocusedObject as ProjectReferenceObject).Name;
}

Непосредственно импорт данных из MS Project в DOCs (где textBoxMSProjectFileName - поле для задания имени файла MS Project, textBoxTargetFolder - наименование/идентификатор целевой папки DOCs для импортируемых данных):

C#
Копировать 
// Очистка лога
msief.Log("");
// Извлечение данных из файла MS Project
msief.ExtractData(textBoxMSProjectFileName.Text);
// Импорт данных
msief.ImportData(new Guid(textBoxTargetFolder.Tag.ToString()));
