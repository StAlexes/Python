using System.Linq;
using System.Threading;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.UI.Client.Common.ViewModels;
using TFlex.DOCs.UI.Client.Common.ViewModels.References;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        if (Context is not UIMacroContext uiMacroContext) // Проверяем, выполняется ли макрос в графическом интерфейсе пользователя
            return;

        var explorerViewModel = uiMacroContext.OwnerViewModel as ReferenceExplorerViewModel; // Получаем модель представления справочника
        // Если ownerViewModel не является моделью представления справочника, выполняем поиск среди всех родителей модели представления
        explorerViewModel ??= uiMacroContext.OwnerViewModel.FindParent<ReferenceExplorerViewModel>();

        if (explorerViewModel is null)
            return;

        var settings = explorerViewModel.Settings; // Получаем настройки        
        var views = settings?.Views?.ToArray(); // Получаем список видов из настроек

        if (views is null)
            return;

        var currentView = settings.CurrentView; // Получаем текущий вид

        if (currentView is null)
            return;

        string name = currentView.Name; // Получаем наименование текущего вида

        // Применяем другой вид из имеющихся
        var appliedView = views.FirstOrDefault(v => v.Id != currentView.Id);

        // Применение вида должно выполняться в UI-потоке
        if (appliedView is not null)
            uiMacroContext.RunOnUIThread(() => _ = explorerViewModel.ApplyView(appliedView, CancellationToken.None));
    }
}
