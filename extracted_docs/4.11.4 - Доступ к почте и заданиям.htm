using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Mail;
using TFlex.DOCs.Model.References.Users;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public void ЗадатьДоступКПочтовымЭлементам()
    {
        // Получаем пользователя 1, к почте и заданиям которого нужно изменить доступ
        var user = Context.Connection.References.Users.Find(new Guid("eadbf357-dc48-4199-96fb-f1e5f4886748")) as User;
        // Получаем пользователя 2, которому нужно предоставить доступ к почте пользователя 1
        var user2 = Context.Connection.References.Users.Find(new Guid("de7e295e-303d-43b2-8486-f39da3d4eee0")) as User;
        // Получаем группу пользователей, которой нужно предоставить доступ к заданиям пользователя 1
        var userGroup = Context.Connection.References.Users.Find(new Guid("23652bb9-627c-41df-a285-876202e69e15")) as UsersGroup;

        if (user is null || user2 is null || userGroup is null)
            return;

        var mailAccessManager = MailAccessManager.GetManager(user, MailAccessType.Messages); // Получаем менеджер почтовых доступов для почты пользователя 1

        mailAccessManager.SetMessagesAccess(user2, MailMessagesAccess.All); // Устанавливаем полный доступ к почте для пользователя 2
        // mailAccessManager.RemoveAccess(user2); // Убираем доступ

        mailAccessManager.Save(); // Сохраняем изменения

        var tasksAccessManager = MailAccessManager.GetManager(user, MailAccessType.Tasks); // Получаем менеджер почтовых доступов для заданий пользователя 1

        tasksAccessManager.SetTasksAccess(userGroup, MailTasksAccess.Read); // Устанавливаем доступ на чтение заданий для группы пользователей
        // tasksAccessManager.RemoveAccess(userGroup); // Убираем доступ

        tasksAccessManager.Save(); // Сохраняем изменения
    }
}
