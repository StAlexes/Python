Подключения
Команды для работы с подключениями в справочниках со сложной иерархией

ТекущееПодключение - возвращает подключение текущего объекта к родительскому

СоздатьПодключение((родительскийОбъект), (дочернийОбъект)) - создает подключение дочернего объекта к родительскому

C#
Копировать 
Подключение подключение = СоздатьПодключение(родительскийОбъект, дочернийОбъект);

 

Подключение.Удалить() - удаляет подключение

Подключение.Изменить() - переводит подключение в состояние редактирования

Подключение.Сохранить() - сохранение изменений

Подключение.ОтменитьИзменения() - отмена изменений

  Внимание

Каждому вызову команды Изменить() должен соответствовать вызов команды Сохранить() или ОтменитьИзменения()

Подключение ["(Параметр подключения)"] - доступ к параметру подключения

C#
Копировать 
подключение["Количество на изделие"] = 5;

 

Подключение.Параметр ["(Параметр подключения)"] - доступ к параметру подключения

C#
Копировать 
подключение.Параметр["Входит в спецификацию"] = false;

 

Подключение.Редактируется - показывает, находится ли подключение в состоянии редактирования

C#
Копировать 
if (!подключение.Редактируется)
    подключение.Изменить();

 

Подключение.РодительскийОбъект - возвращает родительский объект подключения

C#
Копировать 
Объект объект = подключение.РодительскийОбъект;

 

Подключение.ДочернийОбъект - возвращает дочерний объект подключения

C#
Копировать 
Объект объект = подключение.ДочернийОбъект;

 

Объект.ДочерниеПодключения - возвращает список дочерних подключений объекта

C#
Копировать 
foreach (var дочернееПодключение in ТекущийОбъект.ДочерниеПодключения)
{
    дочернееПодключение.Удалить();
}

 

Объект.РодительскиеПодключения - возвращает список родительских подключений объекта

C#
Копировать 
foreach (var родительскоеПодключение in ТекущийОбъект.РодительскиеПодключения)
{
    родительскоеПодключение.Изменить();
    родительскоеПодключение["Количество на изделие"] = 5;
    родительскоеПодключение.Сохранить();
}

 

Объект.СвязанноеПодключение - возвращает связанное подключение

C#
Копировать 
Подключение подключение = ТекущийОбъект.СвязанноеПодключение["Наименование связи"];

 

Объект.СвязанныеПодключения - возвращает связанные подключения

C#
Копировать 
Подключения подключения = ТекущийОбъект.СвязанныеПодключения["Наименование связи"];

 

Объект ["(путь к параметру подключения)"] - возвращает доступ к параметру подключения (доступ к параметрам подключения есть только у объектов, полученных через свойство ДочерниеОбъекты родительского объекта).

C#
Копировать 
Объект деталь = Изделие.ДочерниеОбъекты.First();
int count = деталь["[Подключения].[Количество]"];
