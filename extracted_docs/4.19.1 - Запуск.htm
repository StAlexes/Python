using System;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;
using TFlex.DOCs.References.MasterDataBindings;
using TFlex.DOCs.Synchronization.SyncData;
using TFlex.DOCs.Synchronization.SyncData.MasterServer;

public class Macro(MacroContext context) : MacroProvider(context)
{
    /// <summary>Запуск правила обмена данными между серверами</summary>
    public void RunExchangeWithServer()
    {
        var masterDataBinding = new MasterDataBindingsReference(Context.Connection)
            .Find(new Guid("Guid правила")) as MasterServerExchangeDataRuleReferenceObject; // Находим правило обмена данными

        var masterDataTransfer = masterDataBinding.MasterDataServers.AsList.FirstOrDefault(s => s.Name == "ServerName"); // Находим сервер по наименованию

        // Задаем настройки для сервера, на который будем передавать данные
        var externalData = new MasterServerData(masterDataTransfer)
        {
            MaxAttemps = 3, // Максимальное количество попыток отправки сообщения по сети, в случае потери соединения
            WaitTime = TimeSpan.FromSeconds(5), // Время между попытками отправки сообщения по сети, в случае потери соединения
        };

        // Задаем фильтр для входящих объектов
        var referenceInfo = Context.Connection.ReferenceCatalog.Find(SpecialReference.Units);
        using var filter = new Filter(referenceInfo);
        filter.Terms.AddTerm(referenceInfo.Description[SystemParameterType.ObjectId], ComparisonOperator.GreaterThanOrEqual, 250);

        // Задаем настройки запуска
        var settings = masterDataBinding.CreateDefaultSettings();
        settings.ExchangeDataMode = ExchangeDataMode.Export;
        settings.AddFilter(filter); // Выбираем входящие объекты
        settings.ExternalData = externalData;

        var results = masterDataBinding.Run(settings); // Запускаем
        results.ThrowIfCancelled();
    }

    /// <summary>Запуск правила обмена данными с Xml</summary>
    public void RunExchangeWithXml()
    {
        var диалог = СоздатьДиалогСохраненияФайла();
        диалог.Фильтр = "Файл Xml|*.xml";

        if (!диалог.Показать())
            return;

        var masterDataBinding = new MasterDataBindingsReference(Context.Connection).Find(new Guid("Guid правила")); // Находим правило обмена данными

        // Находим входящие объекты по идентификаторам
        var referenceInfo = Context.Connection.ReferenceCatalog.Find(SpecialReference.Nomenclature);
        var objects = referenceInfo.CreateReference().Find([5, 6]);

        // Задаем настройки запуска
        var settings = masterDataBinding.CreateDefaultSettings();
        settings.ExchangeDataMode = ExchangeDataMode.Export;
        settings.ExternalFilePath = диалог.ИмяФайла; // Путь до файла, куда сохраняем
        settings.AddObjects(objects); // Выбираем входящие объекты
        settings.AdvancedLogging = true; // Ведение журналирования об обмене данными с записью в файл
        settings.EntityLoading = false; // Загружать объекты источника в упрощённом виде
        settings.AllowReferenceObjectProcessingList = true; // Ведение списка обработанных объектов справочника

        var results = masterDataBinding.Run(settings); // Запускаем
        results.ThrowIfCancelled();

        Сообщение(
            results.FilePath,
            $"Длительность выполнения: {results.Duration}" + Environment.NewLine
            + "Обработанные объекты:" + Environment.NewLine
            + String.Join(Environment.NewLine, results.ReferenceObjectActionList.Select(a => $"[{a.Action}] {a.DataObject.SourceObject}"))
            );
    }
}
