Перемещение файлов из текущей папки

В данном примере рассмотрен способ перемещения файлов в текущей папке

Пример перемещения файлов в текущей папке
C#
Копировать 
using System;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Files;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        if (Context.ReferenceObject is not FolderObject sourceFolder) // Проверяем, что текущий объект является папкой
            return;

        var fileReference = new FileReference(Context.Connection);

        // Поиск папки, в которую нужно переместить файл
        if (fileReference.Find(/*Здесь указываем Guid папки*/Guid.Empty) is not FolderObject destinationFolder)
            return;

        var children = sourceFolder.Children.AsList;

        foreach (var child in children.ToList())
        {
            MoveTo(child, destinationFolder);
        }
    }

    private void MoveTo(FileReferenceObject fileReferenceObject, FolderObject destinationFolder)
    {
        if (fileReferenceObject.IsFile)
        {
            if (fileReferenceObject is not FileObject file)
                return;

            if (!file.IsCheckedOutByCurrentUser)
            {
                if (file.CanCheckOut)
                    file.CheckOut(false);
                else
                    return;
            }

            file.BeginChanges(false);
            _ = file.SetParent(destinationFolder); // Перемещаем файл  
            file.EndChanges();

            file.CheckIn("Перемещение файла"); // Если необходимо скрыть окно с применением изменений, то можно так:
            // TFlex.DOCs.Model.Desktop.Desktop.CheckIn(file, "Перемещение файла", false);
        }
        else
        {
            var folder = (FolderObject)fileReferenceObject;
            folder.MoveTo(destinationFolder); // Перемещаем папку
        }
    }
}
