Контексты проектирования

Контексты проектирования позволяют пользователям отдельно прорабатывать изменения изделия, не затрагивая существующую структуру, без создания множества ревизий.
По завершении работы в отдельном контексте проектирования изменения можно перенести в основной контекст или отменить.

using System;
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Nomenclature;
using TFlex.DOCs.Model.References.Nomenclature.DesignContexts;
using TFlex.DOCs.References.StructureTypes;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public void ПолучитьКонтекстыПроектирования()
    {
        // Получаем основной контекст проектирования
        DesignContextObject mainContext = Context.Connection.References.DesignContexts.MainContext;

        // Осуществляем поиск контекста проектирования по его наименованию
        DesignContextObject foundedContextByName = Context.Connection.References.DesignContexts.Find("Дополнительный");

        // Осуществляем поиск контекста проектирования по его идентификатору
        DesignContextObject foundedContextById = Context.Connection.References.DesignContexts.Find(2);

        // Создаем новый контекст проектирования
        DesignContextObject createdDesignContext = Context.Connection.References.DesignContexts.CreateReferenceObject();

        // Заполняем наименование
        createdDesignContext.Name.Value = "Дополнительный";

        // Сохраняем изменения 
        createdDesignContext.EndChanges();
    }

    /// <summary>Применяет изменения в контексте у текущего подключения</summary>
    public void ПрименитьИзмененияВКонтексте()
    {
        // Получаем текущее подключение
        var currentHierarchyLink = Context.HierarchyLink;

        if (currentHierarchyLink is null)
            return;

        // Получаем текущий контекст проектирования
        var currentContext = currentHierarchyLink.Reference.ConfigurationSettings.DesignContext;

        // Применяем изменения в контексте для подключения
        currentContext.ApplyDesignContextChanges([currentHierarchyLink]);
    }

    /// <summary>Переносит изменения текущего подключения в другой контекст</summary>
    public void ПеренестиИзмененияВКонтекст()
    {
        var currentHierarchyLink = Context.HierarchyLink; // Получаем текущее подключение

        if (currentHierarchyLink is null)
            return;

        // Находим контекст проектирования в который будет осуществляться перенос
        var otherContext = Context.Connection.References.DesignContexts.Find("Дополнительный");

        otherContext.MoveChanges(currentHierarchyLink); // Переносим изменения в другой контекст
    }

    /// <summary>Удаление изменений текущего подключения в контексте</summary>
    public void УдалитьИзмененияИзКонтекста()
    {
        var currentHierarchyLink = Context.HierarchyLink; // Получаем текущее подключение

        if (currentHierarchyLink is null)
            return;

        // Получаем текущий контекст проектирования
        var currentContext = currentHierarchyLink.Reference.ConfigurationSettings.DesignContext;

        currentContext.DeleteChanges(currentHierarchyLink); // Удаляем изменения в текущем контексте
    }

    public void СоздтьОбъектВКонтексте()
    {
        var reference = Context.Reference; // Получаем текущий справочник

        // Проверяем, поддерживает ли справочник механизм конфигурирования и работу с контекстами проектирования
        if (!reference.ParameterGroup.SupportsConfigurationSettings || !reference.ParameterGroup.SupportsDesignContexts)
            return;

        // Изменения в контексте для объектов поддерживаются только в справочниках с типами иерархии "Дерево" и "Список"
        if (reference.ParameterGroup.HierarchyType == TFlex.DOCs.Model.Structure.ReferenceHierarchyType.Complex)
            return;

        // Осуществляем поиск контекста проектирования по наименованию
        DesignContextObject otherContext = Context.Connection.References.DesignContexts.Find("Дополнительный");

        // Создаем новые настройки конфигурирования на основе настроек справочника
        var newConfigurationSettings = new ConfigurationSettings(reference.ConfigurationSettings)
        {
            DesignContext = otherContext, // Указываем контекст
            ApplyDesignContext = true
        };

        // Меняем настройки конфигурирования справочника для получения подключений в основном контексте.
        // Настройки конфигурирования учитываются в рамках блока using
        using (reference.ChangeAndHoldConfigurationSettings(newConfigurationSettings))
        {
            var createdObject = reference.CreateReferenceObject(); // Создаем объект в контексте проектирования
            createdObject.EndChanges(); // Сохраняем объект
        }
    }

    public void СоздтьПодключениеВКонтексте()
    {
        var reference = Context.Reference; // Получаем текущий справочник

        // Проверяем, поддерживает ли справочник механизм конфигурирования и работу с контекстами проектирования
        if (!reference.ParameterGroup.SupportsConfigurationSettings || !reference.ParameterGroup.SupportsDesignContexts)
            return;

        // Изменения в контексте для подключений поддерживаются в справочниках с типом иерархии "Сложная иерархия"
        if (reference.ParameterGroup.HierarchyType != TFlex.DOCs.Model.Structure.ReferenceHierarchyType.Complex)
            return;

        // Осуществляем поиск контекста проектирования по наименованию
        DesignContextObject otherContext = Context.Connection.References.DesignContexts.Find("Дополнительный");

        // Создаем новые настройки конфигурирования на основе настроек справочника
        var newConfigurationSettings = new ConfigurationSettings(reference.ConfigurationSettings)
        {
            DesignContext = otherContext, // Указываем контекст
            ApplyDesignContext = true
        };

        var parentObject = reference.Find(1); // Находим родительский объект
        var childObject = reference.Find(2); // Находим дочерний объект

        // Меняем настройки конфигурирования справочника для получения подключений в основном контексте.
        // Настройки конфигурирования учитываются в рамках блока using
        using (reference.ChangeAndHoldConfigurationSettings(newConfigurationSettings))
        {
            var createdLink = parentObject.CreateChildLink(childObject); // Создаем объект в контексте проектирования
            createdLink.EndChanges(); // Сохраняем подключение
        }
    }

    /// <summary>Получает дочерние подключения у текущего объекта для разных контекстов проектирования</summary>
    public void ПолучитьДочерниеПодключенияВКонтексте()
    {
        var currentObject = Context.ReferenceObject; // Получаем текущий объект
        var reference = currentObject.Reference; // Получаем справочник у объекта

        // Создаем настройки конфигурирования
        var newConfigurationSettings = new ConfigurationSettings(reference.ConfigurationSettings)
        {
            ApplyDate = true,
            Date = "Сегодня",
            DesignContext = Context.Connection.References.DesignContexts.MainContext, //Указываем основной контекст
            ApplyDesignContext = true
        };

        // Меняем настройки конфигурирования справочника на момент получения подключений, для получения подключения в основном контексте
        using (reference.ChangeAndHoldConfigurationSettings(newConfigurationSettings))
        {
            // Получаем все подключения на сегодняшнюю дату в основном контексте
            var linksInMainContext = currentObject.Children.GetHierarchyLinks();
        }

        // Находим контекст проектирования в системе
        if (Context.Connection.References.DesignContexts.Find("Технологический") is not DesignContextObject notMainContext)
            return;

        //Проверяем что найденный контекст не является основным контекстом
        if (notMainContext == Context.Connection.References.DesignContexts.MainContext)
            return;

        //Указываем контекст проектирования
        newConfigurationSettings.DesignContext = notMainContext;

        //Меняем настройки конфигурирования справочника на момент получения подключений, для получения подключения в неосновном контексте
        using (reference.ChangeAndHoldConfigurationSettings(newConfigurationSettings))
        {
            //Получаем все подключения на сегодняшнюю дату, в неосновном контексте
            var linksInNotMainContext = currentObject.Children.GetHierarchyLinks();
        }
    }

    public void ОбработкаСтатусаИзмененийОбъектаВКонтексте()
    {
        var currentObject = Context.ReferenceObject; // Получаем текущий объект

        if (currentObject is null)
            return;

        // Получаем статус изменений объекта в контексте проектирования
        DesignContextChangeStatus status = currentObject.GetDesignContextStatus();
        // Для подключений имеется аналогичный метод
        // DesignContextChangeStatus status = Context.HierarchyLink.GetDesignContextStatus();

        switch (status)
        {
            case DesignContextChangeStatus.None: // Нет изменений
                break;
            case DesignContextChangeStatus.Added: // Добавление
                break;
            case DesignContextChangeStatus.Deleted: // Удаление
                break;
            case DesignContextChangeStatus.Changing: //Изменение
                break;
            default:
                break;
        }
    }
}
