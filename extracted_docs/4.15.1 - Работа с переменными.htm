Работа с переменными

В данном примере рассмотрена возможность редактирования переменных документа CAD-системы.

C#
Копировать 
using System.IO;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.FilePreview.CADService;

public class WorkWithVariables : MacroProvider
{
    public WorkWithVariables(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        string path = "Чертёж.grb"; // Путь к grb-документу
        CadDocumentProvider provider = CadDocumentProvider.Connect(Context.Connection, ".grb"); // Подключаемся к CAD-системе 

        using CadDocument document = provider.OpenDocument(path, false); // Открываем документ в режиме редактирования.   

        if (document is null) // Проверяем, был ли открыт документ
            Error($"Файл '{Path.GetFileName(path)}' не может быть открыт");

        VariableCollection variables = document.GetVariables(); // Получаем список переменных документа

        // Находим необходимые переменные по наименованию и устанавливаем значения
        foreach (Variable cadVar in variables)
        {
            switch (cadVar.Name)
            {
                case "Ширина_Д1":
                    cadVar.Value = 15.0;
                    break;
                case "Высота_Д1":
                    cadVar.Value = (double)250;
                    break;
            }
        }

        bool result = variables.Save(); // Сохраняем изменения в списке переменных

        document.Close(true); // Закрываем документ, сохраняя его     
    }
}
