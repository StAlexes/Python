Транзакционное сохранение

Транзакционное сохранение гарантирует, что объекты справочника будут сохранены целиком либо операция будет полностью отменена в случае возникновения ошибки.
Такое сохранение можно осуществить при помощи набора объектов справочника для сохранения (ReferenceObjectSaveSet)

Пример работы с набором объектов справочника для сохранения
C#
Копировать 
using System;
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        Reference testReference = Context.Reference;

        var editableObjects = testReference.Find([1, 2]); // Находим существующие объекты справочника
        int createdObjectsNumber = 2;
        List<ReferenceObject> saveableObjects = new List<ReferenceObject>(editableObjects.Count + createdObjectsNumber);
        Guid param = new Guid("Guid параметра");

        // Редактируем найденные объекты
        foreach (var editableObject in editableObjects)
        {
            editableObject.BeginChanges(false); // Переводим объект в состояние редактирования без перезагрузки
            editableObject[param].Value = 1;

            saveableObjects.Add(editableObject);
        }

        // Создаем новые объекты
        for (int i = 0; i < createdObjectsNumber; i++)
        {
            var newObject = testReference.CreateReferenceObject();
            newObject[param].Value = 1;

            saveableObjects.Add(newObject);
        }

        // Создаем набор объектов справочника для атомарного сохранения
        // В случае возникновения исключения using гарантирует вызов Dispose и откат незавершенных изменений
        using var saveSet = new ReferenceObjectSaveSet();
        saveSet.AddRange(saveableObjects); // Добавляем объекты в набор
        saveSet.EndChanges(); // Завершаем редактирование
    }
}
