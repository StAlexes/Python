Пример работы с фильтром (расширенный)

using System;
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Documents;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class FilterExample : MacroProvider
{
    public FilterExample(MacroContext context)
        : base(context)
    { }

    public override void Run()
    {
        Reference documentReference = new DocumentReference(Context.Connection);
        using Filter filter = CreateFilterForDocuments(documentReference.ParameterGroup); // Создаем фильтр

        List<ReferenceObject> result = documentReference.Find(filter); // Осуществляем поиск объектов с помощью созданного фильтра
        // Выводим результат
        Сообщение("Результат работы фильтра",
            String.Join(",\r\n", result.Select(obj => obj[documentReference.ParameterGroup.DefaultVisibleParameter].Value.ToString())));
    }

    private Filter CreateFilterForDocuments(ParameterGroup documentReference)
    {
        // Генерация фильтра для справочника документов, со следующими условиями: 
        // в названии содержится "чертёж" 
        // И имя одного из связанных файлов равно "2D чертёж.grb"
        // ИЛИ имя одного из связанных файлов содержит "3D"

        ParameterGroup linkToFiles = documentReference.OneToManyLinks
            .First(link => link.Guid == EngineeringDocumentFields.File);

        Filter filter = new Filter(documentReference);

        // Условие: в названии содержится "чертёж"
        ReferenceObjectTerm term1 = new ReferenceObjectTerm(filter.Terms);
        term1.Path.AddParameter(documentReference.DefaultVisibleParameter); // Устанавливаем параметр
        term1.Operator = ComparisonOperator.ContainsSubstring; // Устанавливаем оператор сравнения
        term1.Value = "чертёж"; // Устанавливаем значение для оператора сравнения

        // Условие: имя одного из связанных файлов равно "2D чертёж.grb"
        ReferenceObjectTerm term2 = new ReferenceObjectTerm(filter.Terms);
        term2.Path.AddGroup(linkToFiles); // Добавляем связь в путь к параметру
        term2.Path.AddParameter(linkToFiles.SlaveGroup.DefaultVisibleParameter);
        // У параметра можно получить список поддерживаемых операторов сравнения - ParamterInfo.GetComparisonOperators()
        term2.Operator = term2.Path.CurrentItem.Operators[0];
        term2.Value = "2D чертёж.grb";

        // Условие: в имени одного из связанных файлов содержится "3D". Связываем через оператор "ИЛИ"
        filter.Terms.AddTerm(LogicalOperator.Or, linkToFiles.SlaveGroup.DefaultVisibleParameter, ComparisonOperator.ContainsSubstring, "3D", linkToFiles);
        // Альтернативный вариант создания этого же самого условия:
        //ReferenceObjectTerm term3 = new ReferenceObjectTerm(filter.Terms, LogicalOperator.Or);
        //term3.Path.AddGroup(linkToFiles);
        //term3.Path.AddParameter(linkToFiles.SlaveGroup.DefaultVisibleParameter);
        //term3.Operator = ComparisonOperator.ContainsSubstring;
        //term3.Value = "3D";

        // Группируем условия в отдельную группу (другими словами добавляем скобки)
        TermGroup group = filter.Terms.GroupTerms(new Term[] { term1, term2 });

        // Проверяем фильтр на отсутствие ошибок в условиях
        filter.Validate();

        // Результат - filter.ToString():
        // ([Наименование] содержит 'чертёж' И [Файлы]->[Наименование] = '2D чертёж.grb') ИЛИ [Файлы]->[Наименование] содержит '3D' 

        return filter;
    }
}
Пример добавления параметра из конкретной группы параметров в условие фильтра

using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class FilterExample(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        var userReference = new UserReference(Context.Connection); // Создаем экземпляр справочника
        using var filter = new Filter(userReference.ParameterGroup); // Создаем фильтр

        var groupUserParametersGuid = new Guid("1a78cc76-8dd0-4263-98e7-d6525813e89c");
        ParameterGroup groupUserParameters = userReference.ParameterGroup.OneToOneTables.Find(groupUserParametersGuid); // Находим группу параметров "Параметры пользователя"
        ParameterInfo paramShortName = groupUserParameters?.Parameters.Find(User.Fields.ShortName); // Находим параметр "Короткое имя" в группе параметров "Параметры пользователя"

        if (paramShortName is null)
            return;

        string shortName = "Иванов И.И.";

        // Условие: короткое имя пользователя равно "Иванов И.И."
        ReferenceObjectTerm term = new ReferenceObjectTerm(filter.Terms); // Создаем условие
        // Формируем путь к параметру
        term.Path.AddGroup(groupUserParameters); // Добавляем в путь группу параметров
        term.Path.AddParameter(paramShortName); // Добавляем в путь параметр
        term.Operator = ComparisonOperator.Equal; // Задаем оператор сравнения
        term.Value = shortName; // Задаем значение

        // Альтернативный вариант добавления условия
        //filter.Terms.AddTerm(paramShortName, ComparisonOperator.Equal, shortName, groupUserParameters);

        var user = userReference.FindOne(filter) as User; // Находим пользователя
    }
}
