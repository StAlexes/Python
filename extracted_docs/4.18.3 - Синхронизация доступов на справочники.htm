Пример синхронизации доступов на справочники

using System;
using System.Linq;
using TFlex.DOCs.Client;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Access;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.UI.Client.Common.ViewModels;
using TFlex.DOCs.UI.Client.Common.ViewModels.Base;
using TFlex.DOCs.UI.Client.Common.ViewModels.References.SelectionDialogs;

public class AccessesSynchronizationMacro : MacroProvider
{
    public AccessesSynchronizationMacro(MacroContext context) : base(context)
    { }

    public void СинхронизироватьДоступы()
    {
        if (Context is not UIMacroContext uiContext) // Проверяем, выполняется ли макрос в графическом интерфейсе пользователя
            return;

        // Выбираем справочник, с которого будем считывать доступы (второй аргумент отвечает за выбор нескольких справочников в диалоге)
        var selectReferenceVM = new SelectReferenceViewModel(Context.Connection, false);
        selectReferenceVM.DialogProperties.Title = "Выбор справочника-источника доступов";

        if (!ApplicationManager.OpenDialog(selectReferenceVM, uiContext.OwnerViewModel as LayoutViewModel))
            return;

        ReferenceInfo sourceReference = selectReferenceVM.SelectedObjects[0]; // Справочник-источник доступов

        selectReferenceVM = new SelectReferenceViewModel(Context.Connection, true); // Выбираем справочники для синхронизации доступов
        selectReferenceVM.DialogProperties.Title = "Выбор справочников для синхронизации доступов";

        if (!ApplicationManager.OpenDialog(selectReferenceVM, uiContext.OwnerViewModel as LayoutViewModel))
            return;

        AccessManager allSourceReferenceAccesses = AccessManager.GetAllReferenceAccesses(sourceReference); // Получаем все доступы исходного справочника
        ReferenceInfo[] targetReferences = selectReferenceVM.SelectedObjects; // Из диалога получаем целевые справочники, в которых будут обновлены доступы

        WaitingDialog.Show("Установка доступов", true);

        foreach (ReferenceInfo reference in targetReferences)
        {
            if (!WaitingDialog.NextStep(String.Format("Установка доступов на справочник '{0}'", reference)))
                return;

            AccessManager allReferenceAccesses = AccessManager.GetAllReferenceAccesses(reference); // Получаем все доступы указанного справочника

            foreach (AccessInfo deletedAccess in allReferenceAccesses.Reverse()) // Удаляем имеющиеся доступы
            {
                allReferenceAccesses.RemoveAccess(deletedAccess.Owner, null, deletedAccess.AccessDirection);
            }

            foreach (AccessInfo sourceReferenceAccess in allSourceReferenceAccesses) // Обрабатываем доступы справочника-источника
            {
                // Добавляем аналогичный доступ в целевой-справочник
                allReferenceAccesses.SetAccess(0, sourceReferenceAccess.Owner, sourceReferenceAccess.Access, null, null,
                     sourceReferenceAccess.CommandType, sourceReferenceAccess.AccessTypeID, sourceReferenceAccess.AccessDirection, null);
            }

            allReferenceAccesses.Save(); // Сохраняем изменения
        }

        WaitingDialog.Hide();
    }
}
