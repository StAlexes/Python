using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Assignments;
using TFlex.DOCs.Model.References.Files;
using TFlex.DOCs.Model.References.Tasks;
using TFlex.DOCs.Model.References.Users;


public class ThemesAndTasksMacro : MacroProvider
{
    public ThemesAndTasksMacro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        const string executorName = "Иванов Иван Иванович";
        const string controllerName = "Сергеев Сергей Сергеевич";
        const string filePath = @"Служебные файлы\Шаблоны спецификаций\Спецификации\ВС ГОСТ 2.106-96 ф 3,3а v12.grb";

        var executor = GetUserByFullName(executorName);
        if (executor == null)
            Error("Исполнитель не найден");

        var controller = GetUserByFullName(controllerName);
        if (controller == null)
            Error("Контролёр не найден");

        // Создаем новое поручение
        var assignmentReference = new AssignmentReference(Context.Connection);
        var assignmentClass = assignmentReference.Classes.AssignmentReferenceObject;
        var assignment = (AssignmentReferenceObject) assignmentReference.CreateReferenceObject(assignmentClass);

        // Заполняем параметры нового поручения
        assignment.EndDate.Value = DateTime.Today.AddDays(5);
        assignment.Executor = executor;
        assignment.Controller = controller;
        assignment.Description.Value = "Это тестовое поручение";

        assignment.EndChanges();

        // Находим файл для подключения к задаче
        var document = GetFileByPath(filePath);

        // Создаем новую задачу
        var tasksReference = new TasksReference(Context.Connection);
        var task = (TasksReferenceObject) tasksReference.CreateReferenceObject(tasksReference.Classes.Task);

        if (document != null)
            task.AddLinkedMaterial(document);

        task.AutomaticCalculation.Value = true; // Устанавливаем автоматический расчет процентов
        task.AppointExecutor(Context.Connection.ClientView.GetUser());
        task.AddLinkedAssignment(assignment);
        task.EndChanges();
    }

    /// <summary>
    /// Получает пользователя по полному имени
    /// </summary>
    /// <param name="fullName">Полное имя пользователя</param>
    /// <returns>Пользователь</returns> 
    private User GetUserByFullName(string fullName)
    {
        var userReference = new UserReference(Context.Connection);
        return userReference.FindUser(fullName);
    }

    /// <summary>
    /// Получает файл по относительному пути
    /// </summary>
    /// <param name="path">Относительный путь к файлу</param>
    /// <returns>Файл</returns>
    private FileObject GetFileByPath(string path)
    {
        var fileReference = new FileReference(Context.Connection);
        return fileReference.FindByRelativePath(path) as FileObject;
    }
}
