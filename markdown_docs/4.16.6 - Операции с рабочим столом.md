В данном примере рассмотрена обработка событий, связанных с операциями рабочего стола

```csharp
using TFlex.DOCs.Model.Desktop;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class DesktopChangedEvent : MacroProvider
{
    public DesktopChangedEvent(MacroContext context)
        : base(context)
    { }

    // Обработчик события сервера
    public override void Run()
    {
        ModelEventArgs modelEventArgs = Context.ModelChangedArgs; // Получаем аргументы события изменения в модели

        if (Context.ModelChangedArgs is not ObjectDesktopChangedEventArgs desktopChangedEventArgs) // Получаем аргументы события операции с рабочим столом объекта справочника
            return;

        DesktopOperationInfo operationInfo = desktopChangedEventArgs.Operation; // Получаем описание операции рабочего стола

        string commant = operationInfo.Comment; // Комментарий
        bool keepCheckedOut = operationInfo.KeepCheckedOut; // Получаем или задаем значение, указывающее, необходимо ли оставить объекты в работе
        DesktopOperationType operationType = operationInfo.Type; // Получаем тип операции
    }
}
```
