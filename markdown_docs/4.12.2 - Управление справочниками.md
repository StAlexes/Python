Пример по работе с журналом событий "Управление справочниками"

В данном примере рассмотрена работа с журналом событий "Управление справочниками".
Данный журнал содержит информацию о таких событиях, как создание и удаление справочников, изменение типов и параметров, и др.
Для получения событий по конкретному справочнику необходимо, чтобы в свойствах данного справочника был активирован флаг "Поддержка журнала".

```csharp
/*
Ссылки:
TFLex.DOCs.Data.dll*/

using System;
using System.Collections.Generic;
using TFlex.DOCs.Model.Logging;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Structure;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        // Менеджер журнала 'Управление справочниками'
        var structureLogManager = new StructureLogManager(Context.Connection);
        List<StructureLog> structureLogEvents = structureLogManager.Load(null); // Загружаем события
        List<long> eventIds = new List<long>(structureLogEvents.Count);

        foreach (StructureLog structureLogEvent in structureLogEvents)
        {
            var eventId = structureLogEvent.EventId; // Идентификатор события
            var eventEventTime = structureLogEvent.EventTime; // Время события (UTC)
            var eventType = structureLogEvent.EventType; // Тип события
            string host = structureLogEvent.Host; // Рабочая станция
            Guid referenceGuid = structureLogEvent.ReferenceGuid; // Guid справочника
            int userId = structureLogEvent.UserId; // Идентификатор пользователя

            eventIds.Add(eventId);
        }

        // Получаем детали событий журнала 'Управление справочниками'
        var eventsDetails = structureLogManager.GetStructureLogEventDetailObjects(eventIds.ToArray());

        foreach (var eventDetails in eventsDetails)
        {
            StructureLogBaseLogDescription details = eventDetails.Value;
            switch (details) // В зависимости от типа получаем необходимые данные из детальной информации по событию
            {
                case ReferenceCreateLinkLogDescription referenceCreateLinkLogDescription: // Описание события "Создание связи"
                    ReferencePropertiesLog linkParameters = referenceCreateLinkLogDescription.ParameterProperties;
                    LinkRequired requiredLink = referenceCreateLinkLogDescription.ParameterProperties.LinkRequired; // Правило заполнения связи
                    LinkType linkType = linkParameters.LinkType; // Тип связи
                    break;
                case ReferenceCreateLogDescription referenceCreateLogDescription: // Описание события "Создание справочника"
                    ReferencePropertiesLog referenceParameters = referenceCreateLogDescription.ParameterProperties;
                    bool supportsRecycleBin = referenceParameters.SupportsRecycleBin; // Поддержка корзины
                    ParameterGroupHierarchyType hierarchyType = referenceParameters.HierarchyType; // Тип иерархии
                    break;
                default:
                    continue;
            }
        }
    }
}
```
