Пример работы с фильтрами доступа на справочник

```csharp
using System;
using TFlex.DOCs.Model.Access;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class ReferenceAccessFiltersMacro : MacroProvider
{
    public ReferenceAccessFiltersMacro(MacroContext context) : base(context)
    { }

    public void ОбработатьФильтры()
    {
        var reference = Context.Connection.ReferenceCatalog.Find(new Guid("Guid справочника")); // Находим описание справочника

        if (reference is null)
            return;

        var userObjectAndFilterPairs = new ReferenceAccessFilters(reference); // Получаем информацию о фильтрах доступа на справочник

        foreach (var userObjectAndFilterPair in userObjectAndFilterPairs) // Обрабатываем пары 'Пользователь - фильтр'
        {
            var userObject = userObjectAndFilterPair.Key; // Объект справочника 'Группы и пользователи', для которого задан фильтр
            var filter = userObjectAndFilterPair.Value; // Фильтр
        }

        // Пример добавления фильтра:

        var user = new UserReference(Context.Connection).FindUser("Сотрудник"); // Находим сотрудника из справочника 'Группы и пользователи'

        if (user is not null)
        {
            using Filter newFilter = new Filter(reference.Description); // Создаем фильтр
            newFilter.Terms.AddTerm(reference.Description[SystemParameterType.ObjectId], ComparisonOperator.GreaterThanOrEqual, 5);

            userObjectAndFilterPairs.SetFilter(user, newFilter); // Задаем фильтр для найденного сотрудника
        }

        // userObjectAndFilterPairs.Clear(); // Если необходимо удалить все настроенные фильтры
        bool result = userObjectAndFilterPairs.Save(); // Сохраняем изменения фильтров доступа справочника
    }
}
```
