Справочник "Файлы" предназначен для хранения файлов и работы с ними (при этом физически файлы хранятся на файловом сервере).
В рамках данного справочника API позволяет производить следующие операции над файлами и папками:

- импорт из файловой системы;
- экспорт из файлового хранилища на компьютер пользователя;
- поиск по абсолютному/относительному пути

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using TFlex.DOCs.Model.Desktop;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Files;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        // 1. Добавление файлов из файловой системы
        var fileReference = new FileReference(Context.Connection); // Создаем экземпляр справочника
        var folder = fileReference.FindByRelativePath("Служебные файлы") as FolderObject; // Осуществляем поиск целевой папки по относительному пути

        if (folder is null)
            Error("Внимание!", "Не найдена папка");

        // Настраиваем параметры импорта файлов в справочник "Файлы"
        var importParams = new ImportParameters()
        {
            DestinationFolder = folder, // Задаем папку назначения. Если значение равно null, то импортируемая папка будет добавлена в корень справочника
            Recursive = true, // Выполняем операцию рекурсивно для всех входящих папок
            CreateClasses = true, // Создаем новые типы в случае их отсутствия
            AutoCheckIn = true // Автоматическое применение изменений
        };

        string[] filePaths = [@"C:\Test\t1.txt", @"C:\Test\p1.pdf"];

        IEnumerable<FileObject> fileObjects = fileReference.AddFiles(filePaths, importParams); // Добавляем файлы из файловой системы в справочник "Файлы"
        // Обрабатываем добавленные файлы. Если планируется внесение изменений в файлы,
        // то можно в параметрах импорта выставить флаг AutoCheckIn в false и применить изменения после обработки
        foreach (FileObject fileObject in fileObjects)
        {
        }

        // 2. Теперь рассмотрим импорт папки
        string folderPath = @"C:\FilesToImport"; // Папка в файловой системе - из нее будут копироваться файлы
        fileReference.Import(folderPath, importParams);

        // 3. Загружаем последние версии файлов с сервера в рабочую папку
        FileObject file = fileReference.FindByRelativePath(@"Служебные файлы\Материалы и сортамент\Круг.grb") as FileObject;

        if (file is null)
            return;

        FileReference.GetHeadRevision(new[] { file });

        // 4. Для загрузки файла со снятым атрибутом "Только для чтения" можно воспользоваться следующим кодом:
        file.Export(Path.Combine(Path.GetTempPath(), file.Name), true);

        // 5. Пример создания папки:

        var rootFolder = fileReference.CreateFolder("Новая корневая папка", "Корневая папка"); // Создаем папку в корне справочника
        var subFolder = rootFolder.CreateFolder("Новая вложенная папка", "Вложенная папка"); // Создаем папку внутри корневой папки

        _ = Desktop.CheckIn([rootFolder, subFolder], "Созданы новые папки", false); // Применяем изменения
    }
}
```

Переименование файлов и папок

```csharp
using TFlex.DOCs.Model.Desktop;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Files;

public class RenamingFile : MacroProvider
{
    public RenamingFile(MacroContext context) : base(context) { }

    public override void Run()
    {
        var fileReference = new FileReference(Context.Connection); // Создаем экземпляр справочника "Файлы"
        var folder = fileReference.FindByRelativePath("Корневая папка") as FolderObject; // Находим папку по относительному пути

        if (folder is not null)
        {
            folder.Rename("Переименованная папка"); // Переименовываем папку
            Desktop.CheckIn(folder, "Переименована папка", false); // Применяем изменения
        }

        var file = fileReference.FindByRelativePath(@"Служебные файлы\Библиотека dll\Тестовый файл.PDF") as FileObject; // Находим файл по относительному пути

        if (file is not null)
        {
            if (file.CanCheckOut)
                Desktop.CheckOut(file, false); // Берем на редактирование

            file.Modify(f => f.Name.Value = "Переименованный файл.PDF"); // Переименовываем файл
            Desktop.CheckIn(file, "Переименован файл", false); // Применяем изменения
        }
    }
}
```

Создание дубликата файла (с сохранением подписей)

```csharp
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Files;

public class CreateFileDuplicateMacro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        var fileReference = new FileReference(Context.Connection); // Создаем экземпляр справочника "Файлы"
        var folder = fileReference.FindByRelativePath(@"Дубликаты") as FolderObject; // Находим папку, в которую будем сохранять дубликат
        var file = fileReference.FindByRelativePath(@"Служебные файлы\Библиотека dll\Тестовый файл.PDF") as FileObject; // Находим файл по относительному пути

        if (folder is null || file is null)
            return;

        var fileDuplicate = file.CreateFileDuplicate(folder); // Создаем дубликат файла. Изменения применяются автоматически
    }
}
```
