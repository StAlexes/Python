Команды для работы с отчетами.

`Отчёты.Найти("(Наименование или Guid отчета)", "(Наименование или Guid справочника)")`
- находит отчет по наименованию или Guid'у. Второй аргумент отвечает за справочник, для которого формируется отчет.
Если справочник указан, то он проверяется на соответствие значениям параметра "Справочник" найденных отчетов (в случае, если система нашла несколько отчетов с одинаковым наименованием, вернется первый отчет с соответствующим значением параметра "Справочник")

`Отчёт.Сформировать()`
- формирует отчет для справочника.

`Отчёт.Сформировать("(Объект)")`
- формирует отчет для указанного объекта.

`Отчёт.Сформировать("(Объекты)")`
- формирует отчет для указанных объектов.

Пример настройки и формирования отчета
> Внимание Изменения настроек отчета в макросе у найденного объекта типа 'Отчёт', не влияют на основные настройки отчета в справочнике 'Отчёты'.
> Данные настройки являются копией настроек отчета справочника.

```csharp
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Macros.ObjectModel;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context) { }

    public override void Run()
    {
        Отчёт отчет = Отчёты.Найти("Наименование или Guid отчета", "Наименование или Guid справочника"); // Находим отчет

        if (отчет is null) // Если отчет не найден, выходим
            return;

        ПользовательскийДиалог пользовательскийДиалог = отчет.ПользовательскийДиалог; // Получаем пользовательский диалог отчета

        if (пользовательскийДиалог is not null) // Редактируем параметры пользовательского диалога
        {
            пользовательскийДиалог["Параметр"] = "Значение";
            пользовательскийДиалог.Сохранить();
        }

        отчет.ОткрытьФайл = true; // Задаем значение, указывающее, требуется ли открывать файл отчета после создания во внешней программе
        // отчет.ОткрытьФайлВоВкладке = true; // Задаем значение, указывающее, требуется ли открывать файл отчета на отдельной вкладке после генерации

        отчет.ИмяФайлаОтчёта = "ИмяФайлаОтчета"; // задаем имя файла отчета
        отчет.ПутьКПапке = "ПутьКПапке"; // Задаем путь к папке, содержащей отчет

        // РезультатОтчёта содержит данные о результате генерации отчета
        РезультатОтчёта результатОтчёта = отчет.Сформировать(ТекущийОбъект); // Формируем отчет для выбранного объекта справочника

        // РезультатОтчёта результатОтчёта = отчет.Сформировать(ВыбранныеОбъекты); // Формируем отчет для выбранных объектов справочника
        // РезультатОтчёта результатОтчёта = отчет.Сформировать(); // Формируем отчет для справочника

        string путьКФайлу = результатОтчёта.ПутьКФайлу; // Получаем путь к файлу сформированного отчета
        Объект файлОтчёта = результатОтчёта.ФайлОтчёта; // Получаем файл отчёта

        if (результатОтчёта.СодержитОшибки) // Обрабатываем ошибки, возникшие при генерации отчета
        {
            string описаниеОшибок = результатОтчёта.ОписаниеОшибок; // Получаем описание ошибок, возникших при генерации отчета
        }
    }
}
```
