Ниже предствалена информация по библиотекам для работы с Microsoft Office (описания, инструкции по подключению к макросу, примеры)

SDK Open XML

Пакет SDK Open XML упрощает задачу управления пакетами Open XML и базовыми элементами схемы Open XML.
Классы в пакете SDK инкапсулируют распространенные задачи по работе с Office Open XML

Для работы понадобятся следующие пакеты NuGet:
[DocumentFormat.OpenXml](https://www.nuget.org/packages/DocumentFormat.OpenXml)
[DocumentFormat.OpenXml.Framework](https://www.nuget.org/packages/DocumentFormat.OpenXml.Framework)
Их необходимо скачать, открыть любым архиватором, извлечь библиотеки по пути "lib\net46\" и подключить к макросу.
Информация о подключении библиотек к макросу представлена в разделе «Редактирование макроса»

```csharp
using System;
using System.IO;
using System.Linq;
using DocumentFormat.OpenXml.Packaging;
using DocumentFormat.OpenXml.Spreadsheet;
using TFlex.DOCs.Model.Macros;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        // Открываем электронный документ на чтение
        using var fileStream = new FileStream("Путь к xlsx-файлу", FileMode.Open, FileAccess.Read, FileShare.ReadWrite);
        using SpreadsheetDocument doc = SpreadsheetDocument.Open(fileStream, false);

        WorkbookPart workbookPart = doc.WorkbookPart; // Получаем контейнер верхнего уровня для документа Excel
        // Получаем таблицу общих строк (содержит всю необходимую информацию для отображения строк: текст, свойства форматирования)
        SharedStringTable sharedStringTable = workbookPart.GetPartsOfType<SharedStringTablePart>().First().SharedStringTable;
        Worksheet sheet = workbookPart.WorksheetParts.First().Worksheet; // Получаем лист
        var rows = sheet.Descendants<Row>(); // Получаем строки

        foreach (Row row in rows) // Обрабатываем строки
        {
            foreach (Cell cell in row.Elements<Cell>()) // Обрабатываем ячейки
            {
                if (cell.DataType is null) // Если дата или числовое значение
                {
                    string result = cell.CellValue.Text;
                }
                else if (cell.DataType == CellValues.SharedString) // Если строка
                {
                    int ssid = Int32.Parse(cell.CellValue.Text);
                    string result = sharedStringTable.ElementAt(ssid).InnerText;
                }
                else if (cell.DataType.Value == CellValues.Boolean) // Если логическое значение
                {
                    bool result = Convert.ToBoolean(Convert.ToInt16(cell.CellValue.Text));
                }
            }
        }
    }
}
```

Microsoft.Office.Interop

Для работы со сборками Microsoft.Office.Interop, взаимодействующими с компонентами COM, на компьютере должны быть установлены Microsoft Office Excel и Microsoft Office Word

Необходимы следующие пакеты NuGet:
[Microsoft.Office.Interop.Excel](https://www.nuget.org/packages/Microsoft.Office.Interop.Excel)
[Microsoft.Office.Interop.Word](https://www.nuget.org/packages/Microsoft.Office.Interop.Word)
Их необходимо скачать, открыть любым архиватором, извлечь библиотеки по пути "lib\net20\" и подключить к макросу.
Информация о подключении библиотек к макросу представлена в разделе «Редактирование макроса»

```csharp
using System;
using System.Runtime.InteropServices;
using Microsoft.Office.Interop.Excel;
using TFlex.DOCs.Model.Macros;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        Application objExcel = new Application { DisplayAlerts = false }; // создаем экземпляр приложения Microsoft Excel с отключенным выводом сообщений
        Workbook objWorkBook = objExcel.Workbooks.Open("Путь к xlsx-файлу"); // Открываем книгу
        Worksheet objWorkSheet = objExcel.Sheets[1] as Worksheet; // Получаем лист
        Range xlRange = objWorkSheet.UsedRange; // Получаем используемый диапазон листа

        // Получаем количество столбцов и полей
        int rowCount = xlRange.Rows.Count;
        int colCount = xlRange.Columns.Count;

        for (int i = 1; i <= rowCount; i++)
        {
            for (int j = 1; j <= colCount; j++)
            {
                // Если j == 1, то это новая строка

                // Получаем значение ячейки
                var range = xlRange.Cells[i, j] as Range;
                string result = range.Text.ToString();
            }
        }

        // Освобождаем ресурсы
        Marshal.ReleaseComObject(xlRange);
        Marshal.ReleaseComObject(objWorkSheet);

        objWorkBook.Close(false, Type.Missing, Type.Missing); // Закрываем без сохранения
        Marshal.ReleaseComObject(objWorkBook);

        objExcel.Quit();
        Marshal.ReleaseComObject(objExcel);

        GC.Collect();
        GC.WaitForPendingFinalizers();
    }
}
```

DevExpress

Необходимые библиотеки находятся в папке с клиентом T-FLEX DOCs, поэтому достаточно указать их на вкладке "Ссылки"

Microsoft Excel

```csharp
/*
Ссылки:
DevExpress.Docs.v21.2.dll
DevExpress.Office.v21.2.Core.dll
DevExpress.Spreadsheet.v21.2.Core.dll*/

using System;
using System.IO;
using DevExpress.Spreadsheet;
using TFlex.DOCs.Model.Macros;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        // Открываем электронный документ
        using var stream = new FileStream("Путь к xlsx-файлу", FileMode.Open);
        var workbook = new Workbook();

        if (!workbook.LoadDocument(stream, DevExpress.Spreadsheet.DocumentFormat.Xlsx))
            return;

        Worksheet worksheet = workbook.Worksheets[0]; // Получаем лист
        RowCollection rows = worksheet.Rows; // Получаем строки
        ColumnCollection columns = worksheet.Columns; // Получаем колонки

        int lastRowIndex = worksheet.Rows.LastUsedIndex;
        int lastColumnIndex = worksheet.Columns.LastUsedIndex;

        for (int i = 0; i <= lastRowIndex; i++)
        {
            for (int j = 0; j <= lastColumnIndex; j++)
            {
                Cell cell = worksheet.Cells[i, j]; // Получаем ячейку

                if (!cell.HasData)
                    continue;

                // Получаем значение ячейки
                var cellValue = cell.Value;

                if (cellValue.IsNumeric)
                {
                    double result = cellValue.NumericValue;
                }
                else if (cellValue.IsBoolean)
                {
                    bool result = cellValue.BooleanValue;
                }
                else if (cellValue.IsDateTime)
                {
                    DateTime result = cellValue.DateTimeValue;
                }
                else if (cellValue.IsText)
                {
                    string result = cellValue.TextValue;
                }
            }
        }
    }
}
```

Microsoft Word

```csharp
/*
Ссылки:
DevExpress.RichEdit.v21.2.Core.dll
DevExpress.Office.v21.2.Core.dll
DevExpress.Data.v21.2.dll*/

using DevExpress.XtraRichEdit;
using DevExpress.XtraRichEdit.API.Native;
using TFlex.DOCs.Model.Macros;
using DevExpress.Office.Utils;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        using var reDocServer = new RichEditDocumentServer(); // Создаем класс, предоставляющий функции обработки форматированного текста
        // reDocServer.LoadDocument("Путь к docx-файлу"); // Если необходимо загрузить существующий документ

        Document document = reDocServer.Document; // Получаем созданный документ
        Table table = document.Tables.Create(document.Range.Start, 3, 3); // Создаем таблицу

        // Настраиваем таблицу
        table.TableAlignment = TableRowAlignment.Center;
        table.TableLayout = TableLayoutType.Fixed;
        table.PreferredWidthType = WidthType.Fixed;
        table.PreferredWidth = Units.InchesToDocumentsF(4f);

        table.Rows[1].HeightType = HeightType.Exact;
        table.Rows[1].Height = Units.InchesToDocumentsF(0.8f);

        table[1, 1].PreferredWidthType = WidthType.Fixed;
        table[1, 1].PreferredWidth = Units.InchesToDocumentsF(1.5f);

        table.MergeCells(table[1, 1], table[2, 1]);

        reDocServer.SaveDocument("Путь к docx-файлу", DevExpress.XtraRichEdit.DocumentFormat.OpenXml); // Сохраняем документ
    }
}
```
