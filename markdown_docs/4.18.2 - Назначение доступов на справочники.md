using System;
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Client;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Access;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Macros.ObjectModel;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.UI.Client.Common.ViewModels;
using TFlex.DOCs.UI.Client.Common.ViewModels.Base;
using TFlex.DOCs.UI.Client.Common.ViewModels.References.SelectionDialogs;

public class SetAccessMacro : MacroProvider
{
    public SetAccessMacro(MacroContext context) : base(context)
    { }

    /// <summary>Назначает доступ на справочники</summary>
    public void НазначитьДоступНаСправочники()
    {
        if (Context is not UIMacroContext uiContext) // Проверяем, выполняется ли макрос в графическом интерфейсе пользователя
            return;

        // Выбираем справочники, для которых будет назначаться доступ
        var selectReferenceVM = new SelectReferenceViewModel(Context.Connection, true);

        if (!ApplicationManager.OpenDialog(selectReferenceVM, uiContext.OwnerViewModel as LayoutViewModel))
            return;

        List<AccessGroup> allAccesses = AccessGroup.GetGroups(Context.Connection); // Получаем все доступы в системе
        AccessGroup[] referencesAccesses = allAccesses.FindAll(a => a.Type.IsReference).ToArray(); // Получаем доступы для справочника
        AccessGroup[] objectsAccesses = allAccesses.FindAll(a => a.Type.IsObject).ToArray(); // Получаем доступы для объектов справочника

        // Выводим диалог ввода для выбора доступов и владельца прав доступов
        const string fieldReferencesAccess = "Доступ на справочники";
        const string fieldObjectsAccess = "Доступ на объекты";
        const string fieldOwnerAccess = "Доступ для";

        InputDialog inputDialog = CreateInputDialog("Выбор устанавливаемых доступов");
        inputDialog.Height = 160;
        inputDialog.AddSelectFromList(fieldReferencesAccess, null, true, referencesAccesses);
        inputDialog.AddSelectFromList(fieldObjectsAccess, null, true, objectsAccesses);
        inputDialog.AddSelectFromReference(fieldOwnerAccess, "Группы и пользователи");

        if (!inputDialog.Show())
            return;

        // Получаем доступы из диалога
        var referenceAccess = (AccessGroup)inputDialog.GetValue(fieldReferencesAccess);
        var objectsAccess = (AccessGroup)inputDialog.GetValue(fieldObjectsAccess);
        var accessOwner = (UserReferenceObject)inputDialog.GetValue(fieldOwnerAccess);
        // Если необходимо установить доступ для всех пользователей, то значение accessOwner должно быть null

        WaitingDialog.Show("Установка доступов", true);

        ReferenceInfo[] references = selectReferenceVM.SelectedObjects; // Получаем из диалога выбранные справочники

        foreach (ReferenceInfo reference in references)
        {
            if (!WaitingDialog.NextStep(String.Format("Установка доступов на справочник '{0}'", reference)))
                return;

            AccessManager allReferenceAccesses = AccessManager.GetAllReferenceAccesses(reference); // Получаем все установленные доступы указанного справочника

            SetAccess(allReferenceAccesses, accessOwner, referenceAccess, AccessType.Reference); // Задаем доступ для группы пользователей на справочник
            SetAccess(allReferenceAccesses, accessOwner, objectsAccess, AccessType.Object); // Задаем доступ для группы пользователей на объекты справочника
        }

        WaitingDialog.Hide();
    }

    private void SetAccess(AccessManager accessManager, UserReferenceObject accessOwner, AccessGroup accessGroup, AccessType type)
    {
        // Находим доступ для заданной группы пользователей. Если он задан, то обновляем его, в противном случае - создаем
        AccessInfo userAccess = accessManager.FirstOrDefault(a => a.Owner == accessOwner && a.Access != null && a.Access.Type == type);

        if (userAccess is null)
        {
            accessManager.SetAccess(accessOwner, accessGroup);
        }
        else
        {
            // Данная перегрузка используется для обновления доступа у указанного владельца.
            accessManager.SetAccess(0, userAccess.Owner, accessGroup, userAccess.Access, null,
                userAccess.CommandType, userAccess.AccessTypeID, userAccess.AccessDirection, null);
        }

        accessManager.Save(); // Сохраняем изменения
    }
}
