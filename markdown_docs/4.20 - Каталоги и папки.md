Каталоги служат для группировки объектов, позволяя организовать их просмотр удобным образом. Исходная структура объектов справочника при этом не изменяется.
Каталоги состоят из обособленных или объединенных в группы папок.

Пример работы с каталогами, группами папок и папками

```csharp
using System.Linq;
using TFlex.DOCs.Model.Configuration;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Catalogs;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        if (Context.Reference is null)
            return;

        var catalogManager = new CatalogManager(Context.Reference); // Менеджер каталогов

        var catalog = new Catalog(catalogManager) // Создаем новый каталог
        {
            Name = "Новый каталог", // Устанавливаем наименование каталога
            IsPrivate = false, // Делаем каталог публичным
        };

        // Можем осуществить поиск уже существующего каталога по имени или Guid'у
        // var catalog = catalogManager.Find("Имя или Guid каталога").FirstOrDefault();

        SetCatalogAccess(catalog); // Устанавливаем доступ на каталог
        catalog.Save(); // Сохраняем созданный каталог

        var folderGroup = new FolderGroup(catalog) // Создаем группу папок в указанном каталоге
        {
            Name = "Группа папок", // Устанавливаем наименование группы папок
            ShowObjects = false // Указываем, нужно ли для данной папки отображать вложенные объекты
        };

        folderGroup.Save(); // Сохраняем группу папок

        var userFolder = new UserFolder(folderGroup) { Name = "Папка", ShowObjects = true }; // Создаем новую папку в указанной папке каталога
        userFolder.Save(); // Сохраняем новую папку
        userFolder.Add(Context.Reference.Find([1, 2])); // Добавляем объекты в папку

        var searchFolder = new SearchFolder(catalog) { Name = "Папка поиска", ShowObjects = false }; // Создаем новую папку поиска в каталоге

        ParameterGroup parameterGroup = Context.Reference.ParameterGroup; // Получаем группу параметров
        using var filter = new Filter(parameterGroup); // Создаем фильтр для указанной группы параметров
        filter.Terms.AddTerm(parameterGroup.SystemParameters.Find(SystemParameterType.ObjectId), ComparisonOperator.GreaterThan, 0);

        searchFolder.SetFilter(filter); // Задаем фильтр для папки поиска
        searchFolder.Save(); // Сохраняем новую папку поиска
    }

    private void SetCatalogAccess(Catalog catalog)
    {
        if (catalog is null) // Если нет каталога, то выбрасываем исключение
            Error("Каталог не указан");

        // Задаем доступ
        catalog.AccessUsersUseType = ItemListUseType.ListOnly; // ListOnly - каталог будет доступен только для списка, Exclude - каталог будет доступен тем, кто не входит в список

        var userReferenceObjects = new UserReference(Context.Connection).Find([1, 9, 10, 11, 12]).OfType<UserReferenceObject>(); // Находим объекты справочника 'Группы и пользователи'
        catalog.SetAccessUsers(userReferenceObjects); // Устанавливаем доступ для найденных объектов
    }
}
```
