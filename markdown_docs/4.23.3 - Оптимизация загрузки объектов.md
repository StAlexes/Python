Пример оптимизации загрузки объектов

Для справочника можно задать параметры загрузки объектов с помощью свойства
LoadSettings .
Установка параметров загрузки позволяет получить указанные параметры (и объекты по связям) в момент загрузки объектов.
При обращении к указанным параметрам значения извлекаются из памяти и запросы на сервер не осуществляются.
Данный подход эффективен в случае циклической обработки объектов, т.к. без заданных параметров загрузки
при обработке каждого объекта будет происходить обращение к серверу при получении значения параметра или объекта по связи.

> Внимание LoadSettingsпозволяет загружать параметры и связи к 1. Для загрузки связей к N необходимо воспользоваться классомRelationTree

```csharp
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Structure;

public class OptimizationLoadingObjectsMacro : MacroProvider
{
    public OptimizationLoadingObjectsMacro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Создаем экземпляр справочника "Группы и пользователи"
        var userReference = new UserReference(Context.Connection);
        // Указываем загружаемые параметры
        userReference.LoadSettings.AddParameters(User.Fields.EMail, User.Fields.Birthday);
        // Указываем связь, объекты которой будут загружены вместе с объектами справочника "Группы и пользователи"
        userReference.LoadSettings.AddRelation(UserReferenceObject.RelationKeys.UsersToCalendarLink);
        // Выставляем данный флаг при необходимости загрузки бинарных параметров, к примеру - изображений
        userReference.LoadSettings.LoadBinaryParameters = true;

        // Задаем фильтр, при помощи которого получим только объекты порожденные от типа "Пользователь"
        using var filter = new Filter(userReference.ParameterGroup);
        var term = new ReferenceObjectTerm(filter.Terms);
        term.Path.AddParameter(userReference.ParameterGroup[SystemParameterType.Class]);
        term.Operator = ComparisonOperator.IsInheritFrom;
        term.Value = userReference.Classes.UserBaseType;

        var users = userReference.Find(filter).OfType<User>();

        foreach (var user in users)
        {
            var birthday = user.Birthday; // Получаем уже загруженное значение
            var calendar = user.Calendar; // Получаем уже загруженный объект по связи "Календарь"
            // ...
        }
    }
}
```

Пример оптимизации загрузки объектов при использовании фильтра со списком наборов значений для заданных параметров

Класс (Filter) содержит специальную таблицу значений (`Filter.ValuesTable` ), предназначенную для хранения условий фильтра вида:
([Парам1] = 'Значение1' И [Парам2] = 'Значение2' И ...) ИЛИ
 ([Парам1] = 'Значение3' И [Парам2] = 'Значение4' И ...) ИЛИ
 ([Парам1] = 'Значение5' И [Парам2] = 'Значение6' И ...) ИЛИ ...
Для упрощенной работы с таблицей значений предусмотрен метод расширения SetDataTableValues

```csharp
/*
Ссылки
TFlex.DOCs.Common.dll*/

using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.GlobalParameters;
using TFlex.DOCs.Model.Search;
using TFlex.DOCs.Model.Search.Extensions;
using TFlex.DOCs.Model.Structure;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        // В данном примере оптимизируем поиск объектов справочника 'Глобальные параметры' по двум параметрам: "Наименование" и "Категория".
        // Имеется фильтр вида:
        //     ([Наименование] = 'CheckInCommentsRequired' И [Категория] = 'Системные настройки')
        // ИЛИ ([Наименование] = 'Название предприятия'    И [Категория] = 'Данные о предприятии')
        // ИЛИ ([Наименование] = 'Код предприятия'         И [Категория] = 'Данные о предприятии')
        // ИЛИ ([Наименование] = 'MaximalObjectsNumber'    И [Категория] = 'Системные настройки')
        // Для оптимизации, вместо стандартного добавления условий фильтра, воспользуемся методом SetDataTableValues

        // Получаем справочник "Глобальные параметры"
        GlobalParameterReference globalParametersReference = Context.Connection.References.GlobalParameters;

        // Создаем список параметров, из которых формируются условия по 'И'
        var parameterInfoList = new List<ParameterInfo>
        {
            globalParametersReference.ParameterGroup.OneToOneParameters.FindByName("Наименование"),
            globalParametersReference.ParameterGroup.OneToOneParameters.FindByName("Категория")
        };

        // Указываем значения, которые могут принимать параметры в условиях по 'И'.
        // Значения должны идти в том же порядке, в котором добавлены соответствующие параметры.
        // Следовательно, количество значений должно соответствовать количеству параметров

        var values = new List<string[]>()
        {
            new [] { "CheckInCommentsRequired",     "Системные настройки" },
            new [] { "Название предприятия",        "Данные о предприятии" },
            new [] { "Код предприятия",             "Данные о предприятии" },
            new [] { "MaximalObjectsNumber",        "Системные настройки" },
        };

        using var filter = new Filter(globalParametersReference.ParameterGroup); // Создаем фильтр
        filter.SetDataTableValues(parameterInfoList, values); // Заполняем таблицу значений

        var globalParameters = globalParametersReference.Find(filter).OfType<GlobalParameter>(); // Находим объекты справочника 'Глобальные параметры', соответствующие фильтру

        foreach (var globalParameter in globalParameters)
        {
            // Действия над найденными объектами...
        }
    }
}
```
