В данном примере рассмотрена обработка события "Изменение связи объекта", возникающего перед изменением связи

```csharp
using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    {
    }

    public void ИзменениеСвязиОбъекта()
    {
        if (Context.ModelChangedArgs is not ObjectLinkChangingEventArgs linkChangingEventArgs) // Проверяем, что макрос является обработчиком события "Изменение связи объекта"
            return;

        if (linkChangingEventArgs.Link.LinkGroup.Guid == Guid.Empty) // Проверяем идентификатор связи
        {
            if (linkChangingEventArgs.AddedObject is null) // Если добавляемый объект (AddedObject) отсутствует, значит происходит отключение объекта по связи
                Cancel(false); // Отменяем отключение объекта по связи
        }
    }
}
```
