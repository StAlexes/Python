Для редактирования структуры справочников необходимо использовать специальные классы-строители, находящиеся в пространстве имен
`TFlex.DOCs.Model.Structure.Builders`

Создание нового типа с дополнительной группой параметров и связью

```csharp
/*
Ссылки:
TFlex.DOCs.Common.dll*/

using System;
using TFlex.DOCs.Common;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Classes;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;
using TFlex.DOCs.Model.Structure;
using TFlex.DOCs.Model.Structure.Builders;

public class StructureBuilders(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        var referenceBuilder = GetReferenceBuilder(true); // Получаем экземпляр класса для редактирования структуры справочника

        if (referenceBuilder is null)
            return;

        if (referenceBuilder.IsActive) // Если редактируемый справочник активирован, то деактивируем его
            referenceBuilder.Deactivate();

        var newClass = new ClassObjectBuilder(referenceBuilder.ParameterGroup.Classes) { Name = "Новый тип" }; // Создаем новый тип
        newClass.Save(); // Сохраняем новый тип

        ClassObject newClassObject = newClass.Class; // Получаем из класса-строителя типа непосредственно сам тип

        // Создаем новую группу параметров
        var newGroupBuilder = new ParameterGroupBuilder(ParameterGroupType.TableOneToOne, referenceBuilder.ParameterGroup, newClassObject)
        { Name = "Новая группа параметров" };

        newGroupBuilder.Save(); // Сохраняем новую группу параметров

        // Если справочник уже существует и имеется существующая группа параметров, то подключить ее к типу можно следующим образом:
        ParameterGroup pluggableParameterGroup = referenceBuilder.ParameterGroup.GetAllGroups().Find(new Guid("Guid подключаемой группы параметров"));

        if (pluggableParameterGroup is not null)
        {
            newClass.ParameterGroups.Add(pluggableParameterGroup);
            newClass.Save();
        }

        var parameterGroup = newGroupBuilder.ParameterGroup; // Получаем группу из класса-строителя группы параметров
        string newParameterName = "Новый параметр";

        var newParameter = new ParameterInfoBuilder(parameterGroup) // Создаем параметр
        {
            FieldName = Transliterator.TransliterateString(newParameterName), // Вместо автоматически сгенерированного имени поля в БД можно задать имя на английском, либо воспользовавшись транслитерацией
            Name = newParameterName,
            Type = ParameterType.Int,
            IsVisible = true
        };

        newParameter.Save(); // Сохраняем параметр

        var userReference = new UserReference(Context.Connection); // Создаем экземпляр справочника "Группы и пользователи"

        var newLink = new LinkGroupBuilder(referenceBuilder.ParameterGroup, newClassObject) // Создаем связь "Список со списком" (N к N)
        {
            SlaveGroup = userReference.ParameterGroup,
            TableName = "LinkToUsers",
            LinkType = LinkType.ManyToMany,
            Name = "Связь на пользователей"
        };

        newLink.Save();// Сохраняем связь

        referenceBuilder.Save(); // Сохраняем справочник
        referenceBuilder.Activate(); // Активируем справочник
    }

    private ReferenceBuilder GetReferenceBuilder(bool createReference)
    {
        if (createReference) // Создаем справочник
        {
            ReferenceCatalogFolder folder = Context.Connection.ReferenceCatalog.FindFolderByName("Все справочники"); // Находим папку, в которой будет создан справочник

            if (folder is null)
                return null;

            var referenceBuilder = new ReferenceBuilder(folder); // Создаем экземпляр класса для редактирования структуры справочника
            referenceBuilder.Save(); // Сохраняем

            return referenceBuilder;
        }
        else // Редактирование структуры существующего справочника
        {
            ReferenceInfo referenceInfo = Context.Connection.ReferenceCatalog.Find(new Guid("Guid справочника")); // Находим описание справочника
            return referenceInfo is null ? null : new ReferenceBuilder(referenceInfo); // Создаем экземпляр класса для редактирования структуры справочника
        }
    }
}
```
