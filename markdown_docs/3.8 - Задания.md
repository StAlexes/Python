Команды для работы с заданиями

`СоздатьЗадание()`
- Создает задание.

```csharp
using System;
using System.Collections.Generic;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Macros.ObjectModel;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public void СозданиеЗадания()
    {
        Задание задание = СоздатьЗадание();
        задание.Тема = "Тема задания"; // Заголовок
        задание.Текст = "Текст задания"; // Текст задания
        задание.ДатаНачала = DateTime.Now; // Устанавливаем дату и время начала задания
        задание.ДатаЗавершения = DateTime.Today; // Устанавливаем срок завершения задания

        задание.ДобавитьФайлВложения(@"C:\TFLEX\users.txt"); // Добавляем файл вложения
        // bool результатУдаленияВложенногоФайла = задание.УдалитьФайлВложения(@"C:\TFLEX\users.txt"); // Удаляем файл вложения

        // Добавляем объект вложения
        Объект документ = НайтиОбъект("Документы", "[ID]", 5);
        bool результатДобавленияВложения = задание.ДобавитьОбъектВложения(документ);
        // bool результатУдаленияВложения = задание.УдалитьОбъектВложения(документ); // Удаляем объект вложения

        bool результатДобавленияИсполнителя = задание.ДобавитьИсполнителя("Иванов Иван Иванович"); // Добавляем исполнителя по его наименованию пользователя
        _ = задание.ДобавитьИсполнителя(ТекущийПользователь); // Добавляем пользователя в качестве исполнителя
        // bool результатУдаленияИсполнителя = задание.УдалитьИсполнителя(ТекущийПользователь); // Удаляем исполнителя

        задание.Сохранить(); // Сохраняем задание
        задание.Отправить(); // Отправляем задание

        List<string> файлыВложения = задание.ФайлыВложения; // Получаем вложенные файлы
        Объекты объектыВложения = задание.ОбъектыВложения; // Получаем вложенные объекты

        string ссылкаНаЗапись = задание.СсылкаНаЗапись; // Получаем ссылку на запись
        string ссылкаНаСвойства = задание.СсылкаНаСвойства; // Получаем ссылку на свойства
    }
}
```

`Канцелярия.СоздатьЗадание()`
- Создает задание канцелярии.

```csharp
ЗаданиеКанцелярии задание = Канцелярия.СоздатьЗадание();
задание.Тема = "Тема задания"; // Заголовок
задание.Комментарий = "Комментарий задания";
задание.Ответственный = ТекущийПользователь;
задание.Текст = "Текст задания"; // Текст задания
задание.ДатаНачала = DateTime.Now; // Установить дату и время начала задания
задание.ДатаЗавершения = DateTime.Today; // Установить срок завершения задания
задание.ДобавитьФайлВложения(@"C:\TFLEX\users.txt"); // Добавляет файл вложения
задание.ДобавитьИсполнителя("Иванов Иван Иванович"); // Добавляет исполнителя по его наименованию пользователя
задание.ДобавитьОбъектВложения(ТекущийПользователь); // Добавляет объект вложения
задание.Сохранить(); // Сохраняет задание
задание.Отправить(); // Отправляет задание
```
