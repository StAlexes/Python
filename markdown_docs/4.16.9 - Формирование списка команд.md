В примере рассмотрена обработка события, возникающего во время формирования списка команд справочника.
В обработчике имеется возможность получить и отредактировать данный список.

Пример обработчика события "Формирование списка команд".

```csharp
/*
Ссылки:
TFlex.DOCs.IoC.Infrastructure.dll
TFlex.DOCs.UI.Client.dll
TFlex.DOCs.UI.Client.Common.dll*/

using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.UI.Client.Common.ViewModels;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        if (Context is not UIMacroContext uiMacroContext) // Если макрос выполняется вне контекста графического интерфейса пользователя, выходим
            return;

        // Проверяем, что в контексте графического интерфейса имеется информация о командах
        var currentCommands = uiMacroContext.GetCommands();

        if (currentCommands.Count > 0)
        {
            var printAndExportCommand = currentCommands.FirstOrDefault(command => command.Caption == "Печать и экспорт"); // Находим команду по наименованию
            //var printAndExportCommand = currentCommands.FirstOrDefault(command => command is PrintAndExportCommand); // Команду можно найти по типу

            if (printAndExportCommand is not null)
                currentCommands.Remove(printAndExportCommand); // Удаляем из списка команд
        }
    }
}
```
