В данном примере рассмотрена обработка события "Завершение редактирования свойств объекта".
Событие происходит во время завершения редактирования объекта справочника с панели свойств или в момент закрытия диалога свойств.

Пример обработчика события справочника "Завершение редактирования свойств объекта".

```csharp
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Users;

public class ReferenceUIObjectEndEdit : MacroProvider
{
    public ReferenceUIObjectEndEdit(MacroContext context)
        : base(context)
    { }

    // Обработчик события "Завершение редактирования свойств объекта"
    public override void Run()
    {
        if (Context.CancellingChanges) // Если нажата кнопка "Отмена" или диалог закрыт
            return;

        if (Context.ReferenceObject is not User user)
            return;

        if (user.IsNew && user.Email.IsEmpty && !Question("Не указана электронная почта. Продолжить сохранение?"))
            Cancel(false); // Отменяем сохранение объекта без отображения сообщения о прерывании выполнения макроса
    }
}
```
